def generate_dual_trials(
    n_trials: int,
    timing: TaskTiming,
    input_size: int = 8,
    target_rank : int = 1,
    noise: float = 0.1,
    baseline_value: float = 0.5,
):
    n_steps = timing.n_steps
    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    s_sample = slice(int(n_on[0]), int(n_off[0]))
    s_gng    = slice(int(n_on[1]), int(n_off[1]))

    # gng period
    s_gng_ramp    = slice(int(n_on[1]), int(n_on[3]))
    s_dpa_ramp    = slice(int(n_on[0]), int(n_on[3]))

    s_cue    = slice(int(n_on[2]), int(n_off[2]))
    # s_rwd    = slice(int(n_off[2]), int(n_off[2] + (n_on[3] - n_off[2]) /2))
    s_rwd    = slice(int(n_off[2]), int(n_on[3]))
    s_test   = slice(int(n_on[3]), int(n_off[3]))

    specs = []
    for sample in ["A", "B"]:
        for gng in ["none", "go", "nogo"]:
            for test in ["C", "D"]:
                specs.append((sample, gng, test))

    n_types = len(specs)

    reps = int(np.ceil(n_trials / n_types))
    trial_type = torch.arange(n_types).repeat(reps)[:n_trials]
    trial_type = trial_type[torch.randperm(n_trials)]

    sample_code = torch.tensor(
        [0 if s == "A" else 1 for s, _, _ in specs],
        dtype=torch.long,
    )
    gng_code = torch.tensor(
        [0 if g == "none" else 1 if g == "go" else 2 for _, g, _ in specs],
        dtype=torch.long,
    )
    test_code = torch.tensor(
        [0 if t == "C" else 1 for _, _, t in specs],
        dtype=torch.long,
    )

    sample = sample_code[trial_type]
    gng = gng_code[trial_type]
    test = test_code[trial_type]

    idx_A = sample == 0
    idx_B = sample == 1
    idx_C = test == 0
    idx_D = test == 1

    idx_none = gng == 0
    idx_go = gng == 1
    idx_nogo = gng == 2
    idx_gng = idx_go | idx_nogo

    inputs = noise * torch.randn(n_trials, n_steps, input_size)
    targets = torch.zeros(n_trials, n_steps, target_rank)

    # DPA sample.
    inputs[idx_A, s_sample, 0] += 1.0
    inputs[idx_B, s_sample, 1] += 1.0

    # GNG stimulus.
    inputs[idx_go, s_gng, 4] += 1.0
    inputs[idx_nogo, s_gng, 5] += 1.0

    # Cue only on GNG trials.
    inputs[idx_gng, s_cue, 6] += 1.0

    # DPA test.
    inputs[idx_C, s_test, 2] += 1.0
    inputs[idx_D, s_test, 3] += 1.0

    if target_rank>1:
        targets[..., 0] = torch.nan
        targets[:, :n_on[0]] = 0.0

        baseline_value = None

    if baseline_value is not None:
        targets[:, :s_sample.start] = baseline_value

    # DPA final target.
    idx_pair = (idx_A & idx_C) | (idx_B & idx_D)
    targets[idx_pair, s_test.stop:, -1] = 1.0
    targets[~idx_pair, s_test.stop:, -1] = -1.0

    targets[idx_pair, s_test, -1] = torch.nan
    targets[~idx_pair, s_test, -1] = torch.nan

    # allow ramping on readout
    # targets[:, s_gng_ramp, -1] = torch.nan
    targets[:, s_dpa_ramp, -1] = torch.nan

    targets[idx_go, s_cue, -1] = 1.0
    targets[idx_nogo, s_cue, -1] = -1.0

    targets[idx_go, s_rwd, -1] = 1.0
    targets[idx_nogo, s_rwd, -1] = 0.0

    condition_names = np.array([
        f"{s}_{g}_{t}" if g != "none" else f"{s}_{t}"
        for s, g, t in specs
    ])[trial_type.numpy()]

    return inputs, targets, trial_type, condition_names

import torch
from torch.utils.data import TensorDataset, DataLoader, random_split

def train_val_split(X, y, batch_size=32, frac=0.8):

    # 1. Wrap tensors into a dataset
    dataset = TensorDataset(X.float(), y.float())

    # 2. Train/validation split
    n_total = len(dataset)
    n_train = int(frac * n_total)
    n_val = n_total - n_train

    generator = torch.Generator().manual_seed(0)
    train_dataset, val_dataset = random_split(dataset, lengths=[n_train, n_val], generator=generator)

    # 3. Create DataLoaders
    train_loader = DataLoader(train_dataset, batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size, shuffle=False)

    return train_loader, val_loader

import math
import copy
from typing import Any, Optional

import torch
from torch.utils.data import DataLoader


class Optimization:
    """Small PyTorch train/validation loop with optional true parameter freezing."""

    def __init__(
        self,
        model: torch.nn.Module,
        train_loader: DataLoader,
        val_loader: DataLoader,
        criterion: torch.nn.Module,
        optimizer: torch.optim.Optimizer,
        scheduler: Any | None = None,
        grad_clip_norm: float | None = None,
        num_epochs: int = 100,
        warmup_epochs: int = 25,
        stop_loss: float = 0.005,
        max_val_loss: float = 100.0,

        # Backward-compatible name.
        # If provided, these low-rank columns are zero-gradiented and restored.
        zero_low_rank_grad: Any | None = None,

        # Preferred explicit name.
        freeze_low_rank_cols: Any | None = None,

        # Input dimensions to freeze in wi.weight.
        freeze_input_dims: Any | None = None,

        device: str | torch.device | None = None,
        keep_best: bool = False,
        verbose: bool = True,
    ):
        self.model = model
        self.train_loader = train_loader
        self.val_loader = val_loader
        self.criterion = criterion
        self.optimizer = optimizer
        self.num_epochs = num_epochs
        self.warmup_epochs = warmup_epochs
        self.stop_loss = stop_loss
        self.max_val_loss = max_val_loss
        self.scheduler = scheduler
        self.grad_clip_norm = grad_clip_norm
        self.keep_best = keep_best
        self.verbose = verbose

        self.device = torch.device(device) if device is not None else self._infer_device()
        self.model.to(self.device)
        self.model.device = self.device

        self.train_losses: list[float] = []
        self.val_losses: list[float] = []
        self.learning_rates: list[float] = []
        self.learning_rate = self.optimizer.param_groups[0]["lr"]

        self.best_val_loss = float("inf")
        self.best_state_dict: dict | None = None

        # ------------------------------------------------------------
        # Freezing options
        # ------------------------------------------------------------
        if freeze_low_rank_cols is None:
            freeze_low_rank_cols = zero_low_rank_grad

        self.freeze_low_rank_cols = self._normalize_low_rank_cols(freeze_low_rank_cols)
        self.freeze_input_dims = freeze_input_dims

        # Snapshots used to restore exactly after optimizer.step().
        self._frozen_m = None
        self._frozen_n = None
        self._frozen_wi_weight = None

        if self.freeze_low_rank_cols is not None:
            if not hasattr(self.model, "m") or not hasattr(self.model, "n"):
                raise AttributeError("freeze_low_rank_cols requires model.m and model.n.")

            self._frozen_m = self.model.m.detach().clone()
            self._frozen_n = self.model.n.detach().clone()

        if self.freeze_input_dims is not None:
            if not hasattr(self.model, "wi") or self.model.wi is None:
                raise AttributeError("freeze_input_dims requires model.wi.")
            self._frozen_wi_weight = self.model.wi.weight.detach().clone()

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------

    def _infer_device(self) -> torch.device:
        if hasattr(self.model, "device") and isinstance(self.model.device, torch.device):
            return self.model.device
        return next(self.model.parameters()).device

    def _normalize_low_rank_cols(self, cols):
        """
        Normalize low-rank freeze spec.

        Accepts:
            None
            "all"
            int
            list/tuple/range/tensor of ints
        """
        if cols is None:
            return None

        if cols == "all":
            return "all"

        if isinstance(cols, int):
            return [cols]

        if torch.is_tensor(cols):
            return cols.detach().cpu().long().tolist()

        return list(cols)

    def _predict(self, X: torch.Tensor, y: Optional[torch.Tensor] = None) -> torch.Tensor:
        return self.model(X, y)

    # ------------------------------------------------------------------
    # Gradient zeroing
    # ------------------------------------------------------------------

    def _zero_low_rank_grads(self) -> None:
        if self.freeze_low_rank_cols is None:
            return

        for param in (self.model.m, self.model.n):
            if param.grad is None:
                continue

            if self.freeze_low_rank_cols == "all":
                param.grad.zero_()
            else:
                param.grad[:, self.freeze_low_rank_cols] = 0.0

    def _zero_input_grads(self) -> None:
        if self.freeze_input_dims is None:
            return

        if self.model.wi.weight.grad is not None:
            self.model.wi.weight.grad[:, self.freeze_input_dims] = 0.0

    # ------------------------------------------------------------------
    # True post-step restoration
    # ------------------------------------------------------------------

    def _restore_frozen_low_rank_weights(self) -> None:
        if self.freeze_low_rank_cols is None:
            return

        with torch.no_grad():
            if self.freeze_low_rank_cols == "all":
                self.model.m.copy_(self._frozen_m)
                self.model.n.copy_(self._frozen_n)
            else:
                cols = self.freeze_low_rank_cols
                self.model.m[:, cols] = self._frozen_m[:, cols]
                self.model.n[:, cols] = self._frozen_n[:, cols]

    def _restore_frozen_input_weights(self) -> None:
        if self.freeze_input_dims is None:
            return

        with torch.no_grad():
            self.model.wi.weight[:, self.freeze_input_dims] = (
                self._frozen_wi_weight[:, self.freeze_input_dims]
            )

    def _restore_frozen_weights(self) -> None:
        self._restore_frozen_low_rank_weights()
        self._restore_frozen_input_weights()

    # ------------------------------------------------------------------
    # Epoch loop
    # ------------------------------------------------------------------

    def _run_epoch(self, loader: DataLoader, train: bool) -> float:
        self.model.train(train)

        total_loss = 0.0
        total_examples = 0

        context = torch.enable_grad() if train else torch.no_grad()

        with context:
            for X, y in loader:
                X = X.to(self.device, non_blocking=True)
                y = y.to(self.device, non_blocking=True)

                batch_size = X.size(0)

                if train:
                    self.optimizer.zero_grad(set_to_none=True)

                y_pred = self._predict(X, y)
                loss = self.criterion(y_pred, y)

                if not torch.isfinite(loss):
                    return float("nan")

                if train:
                    loss.backward()

                    # Remove current gradients on frozen parts.
                    self._zero_low_rank_grads()
                    self._zero_input_grads()

                    if self.grad_clip_norm is not None:
                        torch.nn.utils.clip_grad_norm_(
                            self.model.parameters(),
                            self.grad_clip_norm,
                        )

                    self.optimizer.step()

                    # Important: undo AdamW weight decay / optimizer-state drift.
                    self._restore_frozen_weights()

                total_loss += loss.item() * batch_size
                total_examples += batch_size

        if total_examples == 0:
            raise ValueError("Dataloader produced zero examples.")

        return total_loss / total_examples

    # ------------------------------------------------------------------
    # Scheduler / LR
    # ------------------------------------------------------------------

    def _step_scheduler(self, val_loss: float) -> None:
        if self.scheduler is None:
            return

        if isinstance(self.scheduler, torch.optim.lr_scheduler.ReduceLROnPlateau):
            self.scheduler.step(val_loss)
        else:
            self.scheduler.step()

    def _current_lr(self) -> float:
        return self.optimizer.param_groups[0]["lr"]

    # ------------------------------------------------------------------
    # Fit
    # ------------------------------------------------------------------

    def fit(self) -> tuple[list[float], list[float], list[float]]:
        self.train_losses.clear()
        self.val_losses.clear()
        self.learning_rates.clear()

        self.best_val_loss = float("inf")
        self.best_state_dict = None

        for epoch in range(1, self.num_epochs + 1):
            train_loss = self._run_epoch(self.train_loader, train=True)
            val_loss = self._run_epoch(self.val_loader, train=False)

            if epoch >= self.warmup_epochs:
                self._step_scheduler(val_loss)

            self.learning_rate = self._current_lr()

            self.train_losses.append(train_loss)
            self.val_losses.append(val_loss)
            self.learning_rates.append(self.learning_rate)

            if (
                self.keep_best
                and math.isfinite(val_loss)
                and val_loss < self.best_val_loss
            ):
                self.best_val_loss = val_loss
                self.best_state_dict = copy.deepcopy(self.model.state_dict())

            if self.verbose and (epoch % 5 == 0):
                print(
                    f"Epoch {epoch:03d}/{self.num_epochs} | "
                    f"lr: {self.learning_rate:.5f} | "
                    f"train: {train_loss:.4f} | val: {val_loss:.4f}"
                )

            if not math.isfinite(train_loss) or not math.isfinite(val_loss):
                print("Stopping: non-finite loss.")
                break

            if train_loss < self.stop_loss and val_loss < self.stop_loss:
                print(f"Stopping: losses below {self.stop_loss}.")
                break

            if val_loss > self.max_val_loss:
                print(f"Stopping: validation loss above {self.max_val_loss}.")
                break

        if self.keep_best and self.best_state_dict is not None:
            self.model.load_state_dict(self.best_state_dict)

            # After restoring best, enforce frozen weights again.
            self._restore_frozen_weights()

        return self.train_losses, self.val_losses, self.learning_rates

class MaskedInternalReadoutLoss(nn.Module):
    def __init__(
        self,
        readout_index: int = -1,
        criterion: nn.Module | None = None,
        target_weight: float = 1.0,
        target2_weight: float = 1.0,
        zero_weight: float = 1.0,
        bl_weight: float = 0.1,
        baseline_value: float = 0.5,
        baseline_all_channels: bool = True,
    ):
        super().__init__()
        self.readout_index = readout_index
        self.criterion = criterion or nn.MSELoss(reduction="none")

        self.target_weight = target_weight
        self.target2_weight = target2_weight

        self.zero_weight = zero_weight
        self.bl_weight = bl_weight
        self.baseline_value = baseline_value
        self.baseline_all_channels = baseline_all_channels

    @staticmethod
    def masked_mean(loss, mask):
        mask = mask.to(dtype=loss.dtype)
        return (loss * mask).sum() / mask.sum().clamp_min(1.0)

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
        target = y[..., -1]
        pred = y_pred[..., self.readout_index]

        finite = torch.isfinite(pred) & torch.isfinite(target)

        safe_target = torch.where(finite, target, torch.zeros_like(target))
        safe_pred = torch.where(finite, pred, torch.zeros_like(pred))

        zero_mask = finite & (safe_target== 0)
        target_mask = finite & (safe_target.abs()== 1)
        target2_mask = finite & (safe_target.abs()== 2)

        bl_mask = finite & (safe_target == self.baseline_value)

        zero_loss = self.masked_mean(self.criterion(safe_pred, torch.zeros_like(safe_pred)), zero_mask,)

        target_loss = self.masked_mean(self.criterion(safe_pred, safe_target), target_mask,)

        target2_loss = self.masked_mean(self.criterion(safe_pred, safe_target / 2.0), target2_mask,)

        bl_loss = safe_pred.sum() * 0.0
        for i in range(y_pred.shape[-1]):
            bl_i = self.masked_mean(self.criterion(y_pred[..., i], torch.zeros_like(y_pred[..., i])), bl_mask,)
            bl_loss = bl_loss + bl_i

        return (self.zero_weight * zero_loss + self.target_weight * target_loss + self.target2_weight * target2_loss + self.bl_weight * bl_loss)

class MaskedMultiTargetLoss(nn.Module):
    def __init__(
        self,
        criterion: nn.Module | None = None,
        target_weight: float = 1.0,
        zero_weight: float = 1.0,
    ):
        super().__init__()

        self.criterion = criterion or nn.MSELoss(reduction="none")
        self.target_weight = target_weight
        self.zero_weight = zero_weight

    @staticmethod
    def masked_mean(loss, mask):
        mask = mask.to(dtype=loss.dtype)
        return (loss * mask).sum() / mask.sum().clamp_min(1.0)

    def forward_channel(self, pred: torch.Tensor, target: torch.Tensor) -> torch.Tensor:

        finite = torch.isfinite(pred) & torch.isfinite(target)

        safe_target = torch.where(finite, target, torch.zeros_like(target))
        safe_pred = torch.where(finite, pred, torch.zeros_like(pred))

        zero_mask = finite & (safe_target== 0)
        target_mask = finite & (safe_target.abs()== 1)

        zero_loss = self.masked_mean(self.criterion(safe_pred, torch.zeros_like(safe_pred)), zero_mask,)
        target_loss = self.masked_mean(self.criterion(safe_pred, safe_target), target_mask,)

        return (self.zero_weight * zero_loss + self.target_weight * target_loss)

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
        loss = 0.0
        for channel in range(y.shape[-1]):
            pred = y_pred[..., channel].clone()
            target = y[..., channel].clone()

            loss = loss + self.forward_channel(pred, target)
        return loss

class WeightedDualTaskLoss(nn.Module):
    """
    Dual-task loss with separate DPA and GNG weights.

    Assumes generate_dual_trials timing:
        sample: n_on[0] : n_off[0]
        gng:    n_on[1] : n_off[1]
        cue:    n_on[2] : n_off[2]
        test:   n_on[3] : n_off[3]

    DPA loss:
        final decision epoch after test offset: n_off[3] : T

    GNG loss:
        cue/reward epoch: n_on[2] : n_on[3]

    Baseline loss:
        pre-sample epoch: 0 : n_on[0]
    """

    def __init__(
        self,
        timing: TaskTiming,
        readout_index: int = -1,
        criterion: nn.Module | None = None,
        dpa_weight: float = 3.0,
        gng_weight: float = 1.0,
        bl_weight: float = 1.0,
        baseline_value: float = 0.5,
        baseline_all_channels: bool = True,
    ):
        super().__init__()

        self.timing = timing
        self.readout_index = readout_index
        self.criterion = criterion or nn.MSELoss(reduction="none")

        self.dpa_weight = dpa_weight
        self.gng_weight = gng_weight
        self.bl_weight = bl_weight

        self.baseline_value = baseline_value
        self.baseline_all_channels = baseline_all_channels

    @staticmethod
    def masked_mean(loss: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        mask = mask.to(dtype=loss.dtype)
        return (loss * mask).sum() / mask.sum().clamp_min(1.0)

    def forward(self, y_pred: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
        """
        y_pred: [B, T, C_pred]
        y:      [B, T, C_target], NaN = ignore
        """
        B, T, _ = y.shape
        device = y.device

        target = y[..., -1]
        pred = y_pred[..., self.readout_index]

        finite = torch.isfinite(target) & torch.isfinite(pred)

        safe_target = torch.where(finite, target, torch.zeros_like(target))
        safe_pred = torch.where(finite, pred, torch.zeros_like(pred))

        t = torch.arange(T, device=device)

        n_on = self.timing.n_stim_on.to(device)
        n_off = self.timing.n_stim_off.to(device)

        # Epoch masks, shape [T]
        bl_time = t < n_on[0]

        # GNG supervised period: cue + reward, i.e. 6--8 s in your dual timing.
        gng_time = (t >= n_on[2]) & (t < n_on[3])

        # DPA final decision period: after test offset, i.e. 9--10 s.
        dpa_time = t >= n_off[3]

        # Broadcast to [B, T]
        bl_mask = finite & bl_time[None, :] & (safe_target == self.baseline_value)
        gng_mask = finite & gng_time[None, :]
        dpa_mask = finite & dpa_time[None, :]

        # ------------------------------------------------------------
        # Baseline: keep readout, or all internal channels, near zero.
        # ------------------------------------------------------------
        if self.baseline_all_channels:
            bl_mask_all = bl_mask.unsqueeze(-1).expand_as(y_pred)
            bl_loss = self.masked_mean(
                self.criterion(y_pred, torch.zeros_like(y_pred)),
                bl_mask_all,
            )
        else:
            bl_loss = self.masked_mean(
                self.criterion(safe_pred, torch.zeros_like(safe_pred)),
                bl_mask,
            )

        # ------------------------------------------------------------
        # GNG loss: includes Go=1, NoGo=-1 during cue and NoGo=0 during reward.
        # ------------------------------------------------------------
        gng_loss = self.masked_mean(
            self.criterion(safe_pred, safe_target),
            gng_mask,
        )

        # ------------------------------------------------------------
        # DPA loss: final paired/unpaired target after test offset.
        # ------------------------------------------------------------
        dpa_loss = self.masked_mean(
            self.criterion(safe_pred, safe_target),
            dpa_mask,
        )

        return (
            self.dpa_weight * dpa_loss
            + self.gng_weight * gng_loss
            + self.bl_weight * bl_loss
        )

class LowRankModel(nn.Module):
    """
    Low-rank RNN with optional rwd-feedback input channel.
    rwd feedback is teacher-forced.
    """

    def __init__(
        self,
        input_size,
        hidden_size,
        output_size,
        rank=1,
        gain=1.0,
        alpha=0.01,
        alpha_rec=0.01,
        train_inputs=True,
        train_outputs=True,
        rwd_channel=-1,
        rwd_thresh=0.5,
        rwd=True,
        noise=0.1,
        rate_dyn=True,
        device="cpu",
    ):
        super().__init__()
        self.device = torch.device(device)

        self.input_size = input_size
        self.hidden_size = hidden_size
        self.rank = rank
        self.gain = torch.tensor(gain, device=self.device)
        # self.gain = nn.Parameter(torch.tensor(gain, device=self.device))

        self.rwd = rwd
        self.rwd_channel = rwd_channel
        self.rwd_thresh = rwd_thresh

        self.noise = noise
        # self.Cij = (1.0 * (torch.rand(hidden_size, hidden_size, device=self.device) <= sparsity)).clamp_(min=0, max=1)

        # input layer
        self.wi = None
        if input_size > 0:
            self.Ai = 1.0
            self.wi = nn.Linear(input_size, hidden_size, device=self.device)
            with torch.no_grad():
                nn.init.normal_(self.wi.weight, mean=0.0, std=1.0)
                nn.init.zeros_(self.wi.bias)

            if not train_inputs:
                self.Ai = nn.Parameter(torch.randn(1, device=self.device))
                self.wi.weight.requires_grad_(False)
                self.wi.bias.requires_grad_(False)

        # low-rank recurrent layer
        self.m = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))
        self.n = nn.Parameter(torch.randn(hidden_size, rank, device=self.device))

        nn.init.normal_(self.m, mean=0.0, std=1.0)
        nn.init.normal_(self.n, mean=0.0, std=1.0)

        # output layer
        self.wo = None
        if output_size > 0:
            self.Ao = 1.0
            self.wo = nn.Linear(hidden_size, output_size, device=self.device)
            with torch.no_grad():
                nn.init.normal_(self.wo.weight, mean=0.0, std=1.0)
                nn.init.zeros_(self.wo.bias)

            if not train_outputs:
                self.Ao = nn.Parameter(torch.randn(1, device=self.device))
                self.wo.weight.requires_grad_(False)
                self.wo.bias.requires_grad_(False)

        self.nonlinearity = torch.tanh
        self.register_buffer("alpha_rec", torch.tensor(float(alpha_rec)))
        self.register_buffer("exp_alpha_rec", torch.exp(-torch.tensor(float(alpha_rec))))

        self.register_buffer("alpha", torch.tensor(float(alpha)))
        self.register_buffer("exp_alpha", torch.exp(-torch.tensor(float(alpha))))

    def get_readout(self, rates, rec_inputs):
        rec = rates @ self.n / self.hidden_size
        if self.wo is not None:
            ext = self.Ao * self.wo(rates)
            return torch.cat((rec, ext), dim=-1)
        return rec

    def update_dynamics(self, ff_inputs, rec_inputs, rates):
        input_drive = self.Ai * self.wi(ff_inputs) if self.wi is not None else 0.0
        hidden = (rates @ self.n) @ self.m.T / self.hidden_size

        # W = self.Cij * (self.m @ self.n.T) / self.hidden_size
        # hidden = rates @ W.T

        rec_inputs = self.exp_alpha_rec * rec_inputs + (1.0 - self.exp_alpha_rec) * hidden
        # rates = self.nonlinearity(( input_drive + self.gain * rec_inputs))
        phi = self.nonlinearity(( input_drive + self.gain * rec_inputs))
        rates = self.exp_alpha * rates + (1.0 - self.exp_alpha) * phi

        return rates, rec_inputs

    def forward(self, ff_inputs, targets=None, ret_rates=False):
        "ff is (B, T, input_size)"
        B, T = ff_inputs.shape[0], ff_inputs.shape[1]
        N = self.hidden_size

        rec_inputs = torch.zeros(B, N, device=self.device)
        rates = torch.zeros(B, N, device=self.device)

        # rec_inputs = torch.randn(B, N, device=self.device)
        # rates = self.nonlinearity((self.gain * rec_inputs))

        readout_list = []
        rates_list = [] if ret_rates else None
        inputs_list = [] if ret_rates else None

        rwd_next = torch.zeros(B, device=self.device, dtype=ff_inputs.dtype)

        for step in range(T):
            x_t = ff_inputs[:, step].clone()

            if torch.any(rwd_next) & self.rwd:
                x_t[:, self.rwd_channel] = x_t[:, self.rwd_channel] + rwd_next

            noise = self.noise * torch.randn(B, N, device=self.device)
            rates, rec_inputs = self.update_dynamics(x_t, rec_inputs + noise, rates)

            readout = self.get_readout(rates, rec_inputs)
            readout_list.append(readout)

            if ret_rates:
                rates_list.append(rates)
                inputs_list.append(rec_inputs)

            if self.rwd:
                if (targets is not None):
                    rwd_mask = (targets[:, step, self.rwd_channel]==1) & (readout[:, self.rwd_channel]>0.5)
                    rwd_next = rwd_mask.to(dtype=ff_inputs.dtype)
                else:
                    rwd_next.zero_()

        readout_list = torch.stack(readout_list, dim=1)
        if ret_rates:
            return (
                readout_list,
                torch.stack(rates_list, dim=1),
                torch.stack(inputs_list, dim=1),
            )
        return readout_list

import warnings
import torch
import torch.nn as nn
from typing import Optional


class LowRankInitializer:
    """
    Initialize a LowRankModel's loadings (m, n, wi, wo) as i.i.d. samples
    from N(0, Sigma) across neurons.

    Loading vector per neuron:
        [ m_1..m_R, n_1..n_R, I_1..I_{D_in}, w_1..w_{D_out} ]
    """

    BLOCKS = ("m", "n", "in", "out")

    def __init__(
        self,
        sigma: Optional[torch.Tensor] = None,
        var_m: float | torch.Tensor = 1.0,
        var_n: float | torch.Tensor = 1.0,
        var_in: float | torch.Tensor = 1.0,
        var_out: float | torch.Tensor = 1.0,
        overlaps: Optional[dict] = None,
        jitter: float = 1e-8,
        zero_biases: bool = True,
        auto_psd: bool = True,
        psd_tol: float = 1e-8,
        psd_warn_threshold: float = 1e-3,
        preserve_diagonal: bool = True,
    ):
        self.sigma = sigma
        self.var_m = var_m
        self.var_n = var_n
        self.var_in = var_in
        self.var_out = var_out
        self.overlaps = overlaps or {}
        self.jitter = jitter
        self.zero_biases = zero_biases
        self.auto_psd = auto_psd
        self.psd_tol = psd_tol
        self.psd_warn_threshold = psd_warn_threshold
        self.preserve_diagonal = preserve_diagonal

    # ------------------------------------------------------------------
    def _block_sizes(self, model) -> dict:
        R = model.rank
        D_in = model.input_size
        D_out = model.wo.weight.shape[0] if model.wo is not None else 0
        return {"m": R, "n": R, "in": D_in, "out": D_out}

    def _block_offsets(self, sizes: dict) -> dict:
        offsets, cur = {}, 0
        for b in self.BLOCKS:
            offsets[b] = cur
            cur += sizes[b]
        return offsets

    def _expand_var(self, v, size: int) -> torch.Tensor:
        if size == 0:
            return torch.zeros(0)
        if torch.is_tensor(v):
            v = v.flatten().float()
            assert v.numel() == size, f"Expected {size} variances, got {v.numel()}"
            return v
        return torch.full((size,), float(v))

    # ------------------------------------------------------------------
    def _project_psd(self, Sigma: torch.Tensor) -> torch.Tensor:
        """
        Project a symmetric matrix onto the PSD cone via eigenvalue clipping.

        If preserve_diagonal=True, the diagonal of the projected matrix is
        rescaled to match the input's diagonal (so user-specified variances
        are preserved up to a final symmetric rescaling).
        """
        # Symmetrize defensively
        S = 0.5 * (Sigma + Sigma.T)
        d_orig = torch.diag(S).clone()

        evals, evecs = torch.linalg.eigh(S)
        evals_clipped = evals.clamp_min(self.psd_tol)
        S_psd = (evecs * evals_clipped) @ evecs.T

        if self.preserve_diagonal:
            d_new = torch.diag(S_psd).clamp_min(self.psd_tol)
            # Rescale so diag(S_psd) == d_orig, preserving correlation structure
            scale = torch.sqrt(d_orig.clamp_min(0.0) / d_new)
            S_psd = S_psd * scale.unsqueeze(0) * scale.unsqueeze(1)
            # Diagonal is now exactly d_orig (up to fp), confirm PSD after rescale
            # (rescaling by diag(s)·M·diag(s) preserves PSDness)

        # Re-symmetrize against fp drift
        S_psd = 0.5 * (S_psd + S_psd.T)
        return S_psd, evals.min().item()

    def _maybe_project(self, Sigma: torch.Tensor) -> torch.Tensor:
        evals = torch.linalg.eigvalsh(Sigma)
        min_ev = float(evals.min())

        if min_ev >= -self.psd_tol:
            return Sigma  # already PSD up to tol

        if not self.auto_psd:
            raise ValueError(
                f"Sigma is not PSD (min eigenvalue = {min_ev:.3e}). "
                "Pass auto_psd=True to project onto the PSD cone, or fix the spec."
            )

        Sigma_psd, _ = self._project_psd(Sigma)

        # Quantify correction
        delta = (Sigma_psd - Sigma)
        rel_change = (
            torch.linalg.matrix_norm(delta, ord="fro")
            / torch.linalg.matrix_norm(Sigma, ord="fro").clamp_min(1e-12)
        ).item()

        msg = (
            f"Sigma was not PSD (min eigval = {min_ev:.3e}); "
            f"projected to nearest PSD (Frobenius rel. change = {rel_change:.3e})."
        )
        if rel_change > self.psd_warn_threshold:
            msg += (
                " Correction is large — consider reducing overlap magnitudes "
                "or increasing the relevant variances."
            )
        warnings.warn(msg, stacklevel=3)

        return Sigma_psd

    # ------------------------------------------------------------------
    def build_sigma(self, model) -> torch.Tensor:
        sizes = self._block_sizes(model)
        offsets = self._block_offsets(sizes)
        P = sum(sizes.values())

        if self.sigma is not None:
            assert self.sigma.shape == (P, P), (
                f"sigma must be ({P},{P}), got {tuple(self.sigma.shape)}"
            )
            return self._maybe_project(self.sigma.float().clone())

        diag = torch.cat([
            self._expand_var(self.var_m,   sizes["m"]),
            self._expand_var(self.var_n,   sizes["n"]),
            self._expand_var(self.var_in,  sizes["in"]),
            self._expand_var(self.var_out, sizes["out"]),
        ])
        Sigma = torch.diag(diag)

        for ((ba, ia), (bb, ib)), cov in self.overlaps.items():
            assert ba in self.BLOCKS and bb in self.BLOCKS, (
                f"Unknown block: {ba} or {bb}; must be in {self.BLOCKS}"
            )
            assert 0 <= ia < sizes[ba], f"Index {ia} out of range for block {ba}"
            assert 0 <= ib < sizes[bb], f"Index {ib} out of range for block {bb}"
            a = offsets[ba] + ia
            b = offsets[bb] + ib
            Sigma[a, b] = cov
            Sigma[b, a] = cov

        return self._maybe_project(Sigma)

    # ------------------------------------------------------------------
    def _sample_loadings(
        self, Sigma: torch.Tensor, N: int, device, generator=None
    ) -> torch.Tensor:
        P = Sigma.shape[0]

        Sigma = Sigma.to(device)
        Sigma_j = Sigma + self.jitter * torch.eye(
            P, device=device, dtype=Sigma.dtype
        )

        try:
            L = torch.linalg.cholesky(Sigma_j)
        except RuntimeError as e:
            warnings.warn(
                f"Cholesky failed even after PSD projection ({e}); "
                "falling back to eigendecomposition."
            )
            evals, evecs = torch.linalg.eigh(Sigma_j)
            evals = evals.clamp_min(0.0)
            L = evecs * evals.sqrt().unsqueeze(0)

        z = torch.randn(
            N, P,
            device=device,
            dtype=Sigma.dtype,
            generator=generator,
        )

        return z @ L.T

    @torch.no_grad()
    def apply(self, model: nn.Module, seed: Optional[int] = None) -> torch.Tensor:
        Sigma = self.build_sigma(model)
        sizes = self._block_sizes(model)
        offsets = self._block_offsets(sizes)
        N = model.hidden_size
        device = model.device

        gen = None
        if seed is not None:
            gen = torch.Generator(device=device)
            gen.manual_seed(seed)

        X = self._sample_loadings(Sigma, N, device, generator=gen)

        def slc(b):
            return slice(offsets[b], offsets[b] + sizes[b])

        model.m.copy_(X[:, slc("m")])
        model.n.copy_(X[:, slc("n")])

        if sizes["in"] > 0 and model.wi is not None:
            model.wi.weight.copy_(X[:, slc("in")])
            if self.zero_biases:
                model.wi.bias.zero_()

        if sizes["out"] > 0 and model.wo is not None:
            model.wo.weight.copy_(X[:, slc("out")].T)
            if self.zero_biases:
                model.wo.bias.zero_()

        return Sigma

    # ------------------------------------------------------------------
    def validate(self, model, raise_on_fail: bool = True) -> tuple[bool, float]:
        """Build Sigma without projection and report PSD status."""
        old = self.auto_psd
        self.auto_psd = False
        try:
            try:
                Sigma = self.build_sigma(model)
                min_ev = float(torch.linalg.eigvalsh(Sigma).min())
                return True, min_ev
            except ValueError as e:
                if raise_on_fail:
                    raise
                # Re-extract min eigenvalue
                self.auto_psd = True  # to actually build it
                Sigma = self.build_sigma(model)
                min_ev = float(torch.linalg.eigvalsh(Sigma).min())
                return False, min_ev
        finally:
            self.auto_psd = old

import torch

def set_lowrank_spectral_radius(model, target_radius=0.99):
    """
    Rescale m and n so that the spectral radius of (1-alpha)*I + alpha * (m n^T / N)
    equals target_radius.

    This uses the fact that the non-zero eigenvalues of m n^T / N are those of
    C = n^T m / N.
    """
    with torch.no_grad():
        N = model.hidden_size
        # Current 2x2 matrix controlling eigenvalues
        C = (model.n.T @ model.m) / N   # [R, R]
        eigenvalues = torch.linalg.eigvals(C)
        current_max = eigenvalues.abs().max().item()

        if current_max == 0:
            return  # nothing to scale

        # Desired max eigenvalue of C such that (1-alpha) + alpha * desired_max = target_radius
        alpha_ = model.alpha.item()
        desired_max = (target_radius - (1.0 - alpha_)) / alpha_

        # Scale factor for m and n (multiply both by sqrt(desired_max / current_max)
        # to preserve direction but change magnitude of C.
        scale = (desired_max / current_max) ** 0.5
        model.m.mul_(scale)
        model.n.mul_(scale)

import torch
from typing import Optional, List, Dict, Tuple


BLOCKS = ("m", "n", "in", "out")


class FullMixturePopulationInitializer:
    """
    Mixture-of-Gaussians initializer for low-rank RNN loadings.

    Per neuron:

        v_i = [m_i, n_i, wi_i, wo_i]

    with shape:

        P = R + R + D_in + D_out

    Each neuron is assigned to population k and sampled from

        v_i ~ N(mu_k, Sigma_k)

    This is the most general version. Shared input/output marginals are obtained
    by simply making the in/out blocks of Sigma_k identical across populations.
    """

    def __init__(
        self,
        pop_means: List[torch.Tensor],
        pop_covs: List[torch.Tensor],
        pop_probs: Optional[torch.Tensor] = None,
        jitter: float = 1e-6,
        zero_biases: bool = True,
        auto_psd: bool = True,
        psd_tol: float = 1e-8,
    ):
        assert len(pop_means) == len(pop_covs)

        self.K = len(pop_means)
        self.pop_means = [m.flatten().float() for m in pop_means]
        self.pop_covs = [c.float() for c in pop_covs]

        self.pop_probs = (
            pop_probs.float()
            if pop_probs is not None
            else torch.ones(self.K) / self.K
        )
        self.pop_probs = self.pop_probs / self.pop_probs.sum()

        self.jitter = jitter
        self.zero_biases = zero_biases
        self.auto_psd = auto_psd
        self.psd_tol = psd_tol

        P = self.pop_means[0].numel()
        for k, (mu, S) in enumerate(zip(self.pop_means, self.pop_covs)):
            assert mu.numel() == P, f"pop {k}: mean dim {mu.numel()} != {P}"
            assert S.shape == (P, P), (
                f"pop {k}: cov shape {tuple(S.shape)} != {(P, P)}"
            )

        self.P = P

    # ------------------------------------------------------------------
    @staticmethod
    def block_sizes(model) -> Dict[str, int]:
        R = model.rank
        D_in = model.input_size
        D_out = model.wo.weight.shape[0] if model.wo is not None else 0

        return {
            "m": R,
            "n": R,
            "in": D_in,
            "out": D_out,
        }

    @staticmethod
    def block_offsets(sizes: Dict[str, int]) -> Dict[str, int]:
        offsets = {}
        cur = 0
        for b in BLOCKS:
            offsets[b] = cur
            cur += sizes[b]
        return offsets

    @staticmethod
    def loading_dim(sizes: Dict[str, int]) -> int:
        return sum(sizes.values())

    @staticmethod
    def block_slice(block: str, sizes: Dict[str, int]) -> slice:
        offsets = FullMixturePopulationInitializer.block_offsets(sizes)
        return slice(offsets[block], offsets[block] + sizes[block])

    def expected_dim(self, model) -> int:
        return self.loading_dim(self.block_sizes(model))

    # ------------------------------------------------------------------
    def _project_psd(self, S: torch.Tensor) -> torch.Tensor:
        S = 0.5 * (S + S.T)
        evals, evecs = torch.linalg.eigh(S)
        evals = evals.clamp_min(self.psd_tol)
        return (evecs * evals) @ evecs.T

    def _maybe_psd(self, S: torch.Tensor, k: int) -> torch.Tensor:
        S = 0.5 * (S + S.T)
        evals = torch.linalg.eigvalsh(S)

        if float(evals.min()) >= -self.psd_tol:
            return S

        if not self.auto_psd:
            raise ValueError(
                f"Sigma[{k}] is not PSD. "
                f"Minimum eigenvalue = {float(evals.min()):.3e}"
            )

        return self._project_psd(S)

    # ------------------------------------------------------------------
    @torch.no_grad()
    def apply(self, model, seed: Optional[int] = None):
        device = model.device
        sizes = self.block_sizes(model)
        N = model.hidden_size

        expected_P = self.expected_dim(model)
        assert self.P == expected_P, (
            f"Initializer loading dim {self.P} != model loading dim {expected_P}"
        )

        gen = (
            torch.Generator(device=device).manual_seed(seed)
            if seed is not None
            else None
        )

        pop_ids = torch.multinomial(
            self.pop_probs.to(device),
            N,
            replacement=True,
            generator=gen,
        )

        V = torch.zeros(N, self.P, device=device)

        eye = torch.eye(self.P, device=device)

        for k in range(self.K):
            idx = pop_ids == k
            n_k = int(idx.sum())

            if n_k == 0:
                continue

            mu = self.pop_means[k].to(device)
            S = self._maybe_psd(self.pop_covs[k].to(device), k)

            L = torch.linalg.cholesky(S + self.jitter * eye)
            z = torch.randn(n_k, self.P, device=device, generator=gen)

            V[idx] = mu + z @ L.T

        sl_m = self.block_slice("m", sizes)
        sl_n = self.block_slice("n", sizes)
        sl_in = self.block_slice("in", sizes)
        sl_out = self.block_slice("out", sizes)

        model.m.copy_(V[:, sl_m])
        model.n.copy_(V[:, sl_n])

        if sizes["in"] > 0 and model.wi is not None:
            model.wi.weight.copy_(V[:, sl_in])
            if self.zero_biases and model.wi.bias is not None:
                model.wi.bias.zero_()

        if sizes["out"] > 0 and model.wo is not None:
            model.wo.weight.copy_(V[:, sl_out].T)
            if self.zero_biases and model.wo.bias is not None:
                model.wo.bias.zero_()

        return pop_ids

def population_cov_from_correlations(
    spec: Dict[Tuple[str, int, str, int], float],
    sizes: Dict[str, int],
    var_m: float = 1.0,
    var_n: float = 1.0,
    var_in: float = 1.0,
    var_out: float = 1.0,
):
    """
    Build one population covariance matrix Sigma from a correlation spec.

    The full loading vector is ordered as:

        [m_0 ... m_R, n_0 ... n_R, in_0 ... in_D, out_0 ... out_D]

    Parameters
    ----------
    spec:
        Dict mapping

            (block_a, idx_a, block_b, idx_b) -> rho

        where block is one of {"m", "n", "in", "out"}.

    sizes:
        Dict with keys {"m", "n", "in", "out"}.

    var_m, var_n, var_in, var_out:
        Marginal variances for each block.

    Returns
    -------
    mean:
        Zero mean vector of shape [P].

    Sigma:
        Covariance matrix of shape [P, P].
    """

    R = sizes["m"]
    D_in = sizes["in"]
    D_out = sizes["out"]

    offsets = {
        "m": 0,
        "n": R,
        "in": 2 * R,
        "out": 2 * R + D_in,
    }

    block_size = {
        "m": R,
        "n": R,
        "in": D_in,
        "out": D_out,
    }

    block_var = {
        "m": var_m,
        "n": var_n,
        "in": var_in,
        "out": var_out,
    }

    def global_idx(block: str, idx: int) -> int:
        if block not in block_size:
            raise ValueError(f"Unknown block: {block}")

        assert 0 <= idx < block_size[block], (
            f"{block} index {idx} out of range for size {block_size[block]}"
        )

        return offsets[block] + idx

    diag = torch.cat(
        [
            torch.full((R,), float(var_m)),
            torch.full((R,), float(var_n)),
            torch.full((D_in,), float(var_in)),
            torch.full((D_out,), float(var_out)),
        ]
    )

    P = diag.numel()
    Sigma = torch.diag(diag)

    for (ba, ia, bb, ib), rho in spec.items():
        assert -1.0 <= rho <= 1.0, f"rho={rho} outside [-1, 1]"

        ga = global_idx(ba, ia)
        gb = global_idx(bb, ib)

        std_a = block_var[ba] ** 0.5
        std_b = block_var[bb] ** 0.5

        cov = rho * std_a * std_b

        Sigma[ga, gb] = cov
        Sigma[gb, ga] = cov

    mean = torch.zeros(P)

    return mean, Sigma

* Vector fields
** method
The low-rank recurrent state lives approximately on the manifold

\[
h = M \kappa,\qquad M = [m_1,m_2]\in \mathbb{R}^{N\times 2}.
\]

For a fixed feedforward input \(x\), the induced rank-coordinate dynamics are

\[
\kappa^{+}
= (1-\alpha)\kappa + \alpha \Psi(\kappa; x),
\]

where

\[
\Psi(\kappa; x)
= \frac{1}{N} n^\top \tanh(W_i x + b_i + M\kappa).
\]

Thus the phase portrait in the \((\kappa_1,\kappa_2)\) plane is given by the vector field

\[
\Delta \kappa
= \kappa^{+}-\kappa
= \alpha\left[\Psi(\kappa;x)-\kappa\right].
\]

Below I plot the autonomous / delay-period field using \(x=0\). During stimulus epochs the vector field is different because \(x_t\neq 0\).

    if active_dims is not None:

# Low-rank parameter extraction

def low_rank_numpy_params(model):
    """
    Extract parameters for low-rank vector-field calculations.

    Convention:
        h = M κ
        r = tanh(A_i(W_i x + b_i) + gain * h)
        Ψ(κ; x) = n^T r / N
        κ⁺ = κ + β(Ψ(κ; x) - κ)
        β = 1 - exp(-alpha)
    """
    alpha = (
        float(model.alpha.detach().cpu())
        if torch.is_tensor(model.alpha)
        else float(model.alpha)
    )

    if hasattr(model, "_constrained_n"):
        n_eff = model._constrained_n()
    elif hasattr(model, "signed_n"):
        n_eff = model.signed_n()
    else:
        n_eff = model.n

    try:
        Ai = model.Ai.detach().cpu().numpy().astype(np.float64)
    except AttributeError:
        Ai = float(model.Ai) if hasattr(model, "Ai") else 1.0

    if hasattr(model, "gain"):
        gain = (
            float(model.gain.detach().cpu())
            if torch.is_tensor(model.gain)
            else float(model.gain)
        )
    else:
        gain = 1.0

    return {
        "M": model.m.detach().cpu().numpy().astype(np.float64),       # [N, R]
        "Nvec": n_eff.detach().cpu().numpy().astype(np.float64),      # [N, R]
        "Wi": model.wi.weight.detach().cpu().numpy().astype(np.float64),
        "bi": model.wi.bias.detach().cpu().numpy().astype(np.float64),
        "Ai": Ai,
        "gain": gain,
        "beta": 1.0 - np.exp(-alpha),
    }


# ---------------------------------------------------------------------
# Low-rank vector field and Jacobians
# ---------------------------------------------------------------------

def low_rank_field_np(params, kappa, ff_input=None, include_beta=False):
    """
    Vector field in κ-coordinates.

        F(κ; x) = Ψ(κ; x) - κ

    If include_beta=True, returns βF, matching the discrete update increment.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]
    Ai = params["Ai"]
    gain = params["gain"]

    kappa = np.asarray(kappa, dtype=np.float64)
    orig_shape = kappa.shape
    rank = M.shape[1]
    kappa_flat = kappa.reshape(-1, rank)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    input_drive = Ai * (ff_input @ Wi.T + bi)          # [N]
    h = kappa_flat @ M.T                               # [B, N]
    r = np.tanh(input_drive[None, :] + gain * h)       # [B, N]
    psi = r @ Nvec / M.shape[0]                        # [B, R]

    field = psi - kappa_flat

    if include_beta:
        field = params["beta"] * field

    return field.reshape(orig_shape)


def low_rank_jacobian_flow_np(params, kappa, ff_input=None):
    """
    Jacobian of F(κ; x) = Ψ(κ; x) - κ.
    """
    M = params["M"]
    Nvec = params["Nvec"]
    Wi = params["Wi"]
    bi = params["bi"]
    Ai = params["Ai"]
    gain = params["gain"]

    kappa = np.asarray(kappa, dtype=np.float64).reshape(-1)

    if ff_input is None:
        ff_input = np.zeros(Wi.shape[1], dtype=np.float64)
    else:
        ff_input = np.asarray(ff_input, dtype=np.float64)

    input_drive = Ai * (ff_input @ Wi.T + bi)
    u = input_drive + gain * (M @ kappa)

    phi_prime = 1.0 - np.tanh(u) ** 2

    J = Nvec.T @ (phi_prime[:, None] * (gain * M)) / M.shape[0]
    J -= np.eye(M.shape[1])

    return J


def low_rank_jacobian_map_np(params, kappa, ff_input=None):
    """
    Jacobian of the discrete map:

        κ⁺ = κ + βF(κ)
    """
    J_flow = low_rank_jacobian_flow_np(params, kappa, ff_input=ff_input)
    return np.eye(J_flow.shape[0]) + params["beta"] * J_flow


# ---------------------------------------------------------------------
# Projection h -> κ
# ---------------------------------------------------------------------

@torch.no_grad()
def project_rec_inputs_to_kappa(model, rec_inputs):
    """
    Least-squares coordinates for h ≈ Mκ.

    rec_inputs : [..., N]
    returns    : [..., rank]
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    h = torch.as_tensor(rec_inputs, device=device, dtype=dtype)
    orig_shape = h.shape

    hidden_size = model.m.shape[0]
    rank = model.m.shape[1]

    if orig_shape[-1] != hidden_size:
        raise ValueError(f"Expected last dim {hidden_size}, got {orig_shape[-1]}.")

    h_flat = h.reshape(-1, hidden_size)

    sol = torch.linalg.lstsq(model.m, h_flat.T).solution  # [rank, B]
    kappa = sol.T.reshape(*orig_shape[:-1], rank)

    return kappa

@torch.no_grad()
def run_low_rank_with_effective_inputs(
    model,
    ff_inputs,
    targets=None,
    use_reward_feedback=None,
    reward_readout_index=None,
):
    """
    Run a LowRankModel-like network while recording the actual input x_t used.

    This handles online reward feedback if the model uses either

        model.rwd_channel

    or

        model.reward_channel

    and if reward feedback is active.

    Returns
    -------
    readouts    : [B, T, output_size]
    rates       : [B, T, N]
    rec_inputs  : [B, T, N]
    effective_x : [B, T, input_size]
    """
    device = _model_device(model)
    dtype = _model_dtype(model)

    ff_inputs = ff_inputs.to(device=device, dtype=dtype)
    if targets is not None:
        targets = targets.to(device=device)

    batch_size, T, input_size = ff_inputs.shape
    hidden_size = model.m.shape[0]

    rec_inputs = torch.zeros(batch_size, hidden_size, device=device, dtype=dtype)
    rates = torch.zeros(batch_size, hidden_size, device=device, dtype=dtype)

    noise = getattr(model, "noise", 0.0)

    rwd_channel = getattr(model, "rwd_channel", None)
    if rwd_channel is None:
        rwd_channel = getattr(model, "reward_channel", None)

    if use_reward_feedback is None:
        use_reward_feedback = (
            targets is not None
            and rwd_channel is not None
            and (
                bool(getattr(model, "rwd", False))
                or getattr(model, "reward_channel", None) is not None
            )
        )

    readout_list = []
    rates_list = []
    rec_list = []
    eff_input_list = []

    rwd_next = torch.zeros(batch_size, device=device, dtype=dtype)

    for step in range(T):
        x_t = ff_inputs[:, step].clone()

        if use_reward_feedback:
            x_t[:, rwd_channel] = x_t[:, rwd_channel] + rwd_next

        sigma = noise * torch.randn(batch_size, hidden_size, device=device, dtype=dtype)
        rates, rec_inputs = model.update_dynamics(x_t, rec_inputs + sigma, rates)
        readout = model.get_readout(rates, rec_inputs)

        readout_list.append(readout)
        rates_list.append(rates)
        rec_list.append(rec_inputs)
        eff_input_list.append(x_t)

        if use_reward_feedback:
            if reward_readout_index is None:
                if rwd_channel < readout.shape[-1]:
                    idx = rwd_channel
                else:
                    idx = -1
            else:
                idx = reward_readout_index

            rwd_mask = (
                (targets[:, step, rwd_channel] == 1)
                & (readout[:, idx] > 0.5)
            )
            rwd_next = rwd_mask.to(dtype=dtype)
        else:
            rwd_next.zero_()

    return (
        torch.stack(readout_list, dim=1),
        torch.stack(rates_list, dim=1),
        torch.stack(rec_list, dim=1),
        torch.stack(eff_input_list, dim=1),
    )

def auto_limits_from_trajs(kappa_traj, qlo=1, qhi=99, pad=1.25, symmetric=True):
    z = kappa_traj.reshape(-1, kappa_traj.shape[-1])

    lo = np.percentile(z[:, :2], qlo, axis=0)
    hi = np.percentile(z[:, :2], qhi, axis=0)

    if symmetric:
        lim = float(pad * np.max(np.abs(np.concatenate([lo, hi]))))
        lim = max(lim, 1e-6)
        return (-lim, lim), (-lim, lim)

    center = 0.5 * (lo + hi)
    half = 0.5 * (hi - lo) * pad

    return (
        (float(center[0] - half[0]), float(center[0] + half[0])),
        (float(center[1] - half[1]), float(center[1] + half[1])),
    )


def make_vector_field_grid(
    model,
    ff_input,
    xlim,
    ylim,
    n_grid=151,
    include_beta=False,
):
    params = low_rank_numpy_params(model)

    ff_input_np = (
        torch.as_tensor(ff_input)
        .detach()
        .cpu()
        .numpy()
        .astype(np.float64)
    )

    k1 = np.linspace(xlim[0], xlim[1], n_grid)
    k2 = np.linspace(ylim[0], ylim[1], n_grid)

    K1, K2 = np.meshgrid(k1, k2)
    K = np.stack([K1, K2], axis=-1)

    field = low_rank_field_np(
        params,
        K,
        ff_input=ff_input_np,
        include_beta=include_beta,
    )

    U = field[..., 0]
    V = field[..., 1]
    speed = np.sqrt(U**2 + V**2)

    return K1, K2, U, V, speed, params, ff_input_np

def merge_roots(roots, residuals, merge_tol=5e-2):
    if len(roots) == 0:
        return np.empty((0, 2)), np.empty((0,))

    roots = np.asarray(roots, dtype=np.float64)
    residuals = np.asarray(residuals, dtype=np.float64)

    order = np.argsort(residuals)
    kept = []
    kept_res = []

    for idx in order:
        r = roots[idx]
        if all(np.linalg.norm(r - q) > merge_tol for q in kept):
            kept.append(r)
            kept_res.append(residuals[idx])

    return np.stack(kept, axis=0), np.asarray(kept_res)


def find_all_fixed_points(
    model,
    xlim,
    ylim,
    ff_input,
    n_seeds=41,
    residual_tol=1e-8,
    merge_tol=5e-2,
):
    if model.m.shape[1] != 2:
        raise ValueError("Fixed-point finding here assumes rank == 2.")

    params = low_rank_numpy_params(model)

    ff_input_np = (
        torch.as_tensor(ff_input)
        .detach()
        .cpu()
        .numpy()
        .astype(np.float64)
    )

    def fun(k):
        return low_rank_field_np(params, k, ff_input=ff_input_np).reshape(-1)

    def jac(k):
        return low_rank_jacobian_flow_np(params, k, ff_input=ff_input_np)

    k1_seeds = np.linspace(xlim[0], xlim[1], n_seeds)
    k2_seeds = np.linspace(ylim[0], ylim[1], n_seeds)

    roots = []
    residuals = []

    for k1_0 in k1_seeds:
        for k2_0 in k2_seeds:
            sol = root(
                fun,
                np.array([k1_0, k2_0], dtype=np.float64),
                jac=jac,
                method="hybr",
                options={"xtol": 1e-12, "maxfev": 1000},
            )

            if not sol.success:
                continue

            fp = sol.x
            residual = np.linalg.norm(fun(fp))

            in_bounds = (
                xlim[0] <= fp[0] <= xlim[1]
                and ylim[0] <= fp[1] <= ylim[1]
            )

            if residual <= residual_tol and in_bounds:
                roots.append(fp)
                residuals.append(residual)

    return merge_roots(roots, residuals, merge_tol=merge_tol)


def classify_fixed_points(model, fixed_points, ff_input, eig_tol=1e-5):
    params = low_rank_numpy_params(model)

    ff_input_np = (
        torch.as_tensor(ff_input)
        .detach()
        .cpu()
        .numpy()
        .astype(np.float64)
    )

    labels = []
    eigvals = []

    for fp in fixed_points:
        J = low_rank_jacobian_map_np(params, fp, ff_input=ff_input_np)
        ev = np.linalg.eigvals(J)
        rad = np.abs(ev)

        n_stable = np.sum(rad < 1.0 - eig_tol)
        n_unstable = np.sum(rad > 1.0 + eig_tol)

        if n_stable == len(rad):
            label = "attractor"
        elif n_unstable == len(rad):
            label = "repeller"
        elif n_stable > 0 and n_unstable > 0:
            label = "saddle"
        else:
            label = "nonhyperbolic"

        labels.append(label)
        eigvals.append(ev)

    if len(eigvals) == 0:
        return np.asarray([], dtype=object), np.empty((0, 2), dtype=np.complex128)

    return np.asarray(labels, dtype=object), np.stack(eigvals, axis=0)

def _canonical_input_mask(effective_x, dims, threshold=0.35, atol_inactive=0.35):
    """
    Select time points whose effective input matches a frozen-input panel.

    dims=None:
        approximately zero input.

    dims=[k]:
        channel k active and all other channels inactive.
    """
    x = effective_x.detach().cpu()

    if dims is None:
        return x.abs().amax(dim=-1) < atol_inactive

    dims = list(dims)

    active = torch.ones(x.shape[:2], dtype=torch.bool)
    for d in dims:
        active &= x[..., d] > threshold

    inactive_dims = [d for d in range(x.shape[-1]) if d not in dims]
    inactive = x[..., inactive_dims].abs().amax(dim=-1) < atol_inactive

    return active & inactive


def _segments_from_mask(mask_1d, min_len=2):
    idx = np.where(mask_1d)[0]

    if idx.size == 0:
        return []

    breaks = np.where(np.diff(idx) > 1)[0] + 1
    chunks = np.split(idx, breaks)

    out = []
    for c in chunks:
        if len(c) >= min_len:
            out.append(slice(int(c[0]), int(c[-1]) + 1))

    return out

def default_arrow_times(T, n_arrows=4, arrow_step=3):
    if T <= arrow_step + 1:
        return np.array([], dtype=int)

    lo = 1
    hi = T - arrow_step - 1
    n_arrows = min(n_arrows, max(1, hi - lo + 1))

    return np.unique(np.linspace(lo, hi, n_arrows).astype(int))


def add_time_arrows(
    ax,
    xy,
    times=None,
    color="k",
    arrow_step=3,
    mutation_scale=13,
    lw=1.5,
    alpha=0.95,
    zorder=8,
    min_length=1e-4,
):
    xy = np.asarray(xy, dtype=np.float64)
    T = xy.shape[0]

    if times is None:
        times = default_arrow_times(T, n_arrows=4, arrow_step=arrow_step)

    for t in times:
        t = int(t)

        if t < 0:
            t += T

        if t < 0 or t >= T - 1:
            continue

        j0 = t
        j1 = min(t + arrow_step, T - 1)

        while j1 < T and np.linalg.norm(xy[j1] - xy[j0]) < min_length:
            j0 += 1
            j1 = min(j0 + arrow_step, T - 1)

            if j0 >= T - 1:
                break

        if j0 >= T - 1:
            continue

        p0 = xy[j0]
        p1 = xy[j1]

        if np.linalg.norm(p1 - p0) < min_length:
            continue

        ax.annotate(
            "",
            xy=p1,
            xytext=p0,
            arrowprops=dict(
                arrowstyle="-|>",
                color=color,
                lw=lw,
                alpha=alpha,
                shrinkA=0,
                shrinkB=0,
                mutation_scale=mutation_scale,
            ),
            zorder=zorder,
        )

def _parse_dual_name(name):
    """
    Parse dual-task condition names.

    Examples:
        "A_C"        -> sample=A, gng=none, test=C
        "A_go_D"     -> sample=A, gng=go,   test=D
        "B_nogo_C"   -> sample=B, gng=nogo, test=C
    """
    parts = str(name).split("_")

    if len(parts) == 2:
        sample, test = parts
        gng = "none"
    elif len(parts) == 3:
        sample, gng, test = parts
    else:
        raise ValueError(f"Unexpected dual condition name: {name}")

    pair = f"{sample}{test}"

    return {
        "sample": sample,
        "gng": gng,
        "test": test,
        "pair": pair,
    }


def condition_indices_for_task(
    inputs,
    timing,
    task,
    condition_names=None,
    dual_mode="conditions",
):
    """
    Return condition indices and metadata.

    task:
        "dpa", "gng", or "dual"

    dual_mode:
        "conditions" : concrete labels such as A_go_C
        "groups"     : grouped labels AC/AD/BC/BD/Go/NoGo/DPAOnly
    """
    task = task.lower()
    X = inputs.detach().cpu()

    n_on = timing.n_stim_on
    n_off = timing.n_stim_off

    # -----------------------------
    # DPA
    # -----------------------------
    if task == "dpa":
        sample_epoch = slice(int(n_on[0]), int(n_off[0]))
        test_epoch = slice(int(n_on[1]), int(n_off[1]))

        sample_drive = X[:, sample_epoch, 0:2].mean(dim=1)
        test_drive = X[:, test_epoch, 2:4].mean(dim=1)

        is_A = sample_drive[:, 0] > sample_drive[:, 1]
        is_B = ~is_A
        is_C = test_drive[:, 0] > test_drive[:, 1]
        is_D = ~is_C

        cond_idx = {
            "AC": torch.where(is_A & is_C)[0],
            "AD": torch.where(is_A & is_D)[0],
            "BC": torch.where(is_B & is_C)[0],
            "BD": torch.where(is_B & is_D)[0],
        }

        meta = {
            "AC": dict(sample="A", test="C", pair="AC"),
            "AD": dict(sample="A", test="D", pair="AD"),
            "BC": dict(sample="B", test="C", pair="BC"),
            "BD": dict(sample="B", test="D", pair="BD"),
        }

        return cond_idx, meta

    # -----------------------------
    # GNG
    # -----------------------------
    if task == "gng":
        stim_epoch = slice(int(n_on[0]), int(n_off[0]))
        stim_drive = X[:, stim_epoch, 4:6].mean(dim=1)

        is_go = stim_drive[:, 0] > stim_drive[:, 1]
        is_nogo = ~is_go

        cond_idx = {
            "Go": torch.where(is_go)[0],
            "NoGo": torch.where(is_nogo)[0],
        }

        meta = {
            "Go": dict(gng="go"),
            "NoGo": dict(gng="nogo"),
        }

        return cond_idx, meta

    # -----------------------------
    # Dual
    # -----------------------------
    if task == "dual":
        if condition_names is None:
            raise ValueError(
                "For task='dual', pass condition_names from generate_dual_trials."
            )

        names = np.asarray(condition_names).astype(str)

        if dual_mode == "conditions":
            cond_idx = {}
            meta = {}

            for name in sorted(np.unique(names)):
                idx = np.where(names == name)[0]
                cond_idx[name] = torch.as_tensor(idx, dtype=torch.long)
                meta[name] = _parse_dual_name(name)

            return cond_idx, meta

        if dual_mode == "groups":
            groups = {
                "AC": [],
                "AD": [],
                "BC": [],
                "BD": [],
                "Go": [],
                "NoGo": [],
                "DPAOnly": [],
            }

            for i, name in enumerate(names):
                info = _parse_dual_name(name)
                pair = info["pair"]
                gng = info["gng"]

                if pair in groups:
                    groups[pair].append(i)

                if gng == "go":
                    groups["Go"].append(i)
                elif gng == "nogo":
                    groups["NoGo"].append(i)
                elif gng == "none":
                    groups["DPAOnly"].append(i)

            cond_idx = {
                k: torch.as_tensor(v, dtype=torch.long)
                for k, v in groups.items()
            }

            meta = {
                "AC": dict(pair="AC"),
                "AD": dict(pair="AD"),
                "BC": dict(pair="BC"),
                "BD": dict(pair="BD"),
                "Go": dict(gng="go"),
                "NoGo": dict(gng="nogo"),
                "DPAOnly": dict(gng="none"),
            }

            return cond_idx, meta

        raise ValueError(f"Unknown dual_mode: {dual_mode}")

    raise ValueError(f"Unknown task: {task}")

def flow_specs_for_task(
    timing,

    input_size,

    if task == "dpa":

            dict(name="C", dims=[2], conds=["AC", "BC"]),
            dict(name="D", dims=[3], conds=["AD", "BD"]),

            dict(name="NoGo", dims=[5], conds=["NoGo"]),

        ]

    if task == "dual":

            B_conds = [c for c in labels if meta[c]["sample"] == "B"]
            C_conds = [c for c in labels if meta[c]["test"] == "C"]
            D_conds = [c for c in labels if meta[c]["test"] == "D"]
            Go_conds = [c for c in labels if meta[c]["gng"] == "go"]
            NoGo_conds = [c for c in labels if meta[c]["gng"] == "nogo"]

            A_conds = ["AC", "AD"]
            B_conds = ["BC", "BD"]
            C_conds = ["AC", "BC"]
            D_conds = ["AD", "BD"]
            Go_conds = ["Go"]
            NoGo_conds = ["NoGo"]
            Cue_conds = ["Go", "NoGo"]

        return [
            dict(name="Autonomous", dims=None, conds=auto_conds),
            dict(name="A", dims=[0], conds=A_conds),
            dict(name="B", dims=[1], conds=B_conds),

            dict(name="C", dims=[2], conds=C_conds),
            dict(name="D", dims=[3], conds=D_conds),

def plot_task_flow_fields(
    model,
    inputs,
    timing,
    task,
    targets=None,
    condition_names=None,
    dual_mode="conditions",
    xlim=None,
    ylim=None,
    n_grid=151,
    n_fp_seeds=41,
    include_beta_in_field=False,
    qlo=1,
    qhi=99,
    pad=1.25,
    symmetric_auto=True,
    speed_percentile=98,
    figsize_per_panel=4.2,
    input_threshold=0.35,
    inactive_atol=0.35,
    show_single_trials=False,
    max_single_trials=12,
    max_autonomous_conditions=None,
):
    """
    Generic low-rank phase portrait for DPA, GNG, and Dual tasks.

    Important behavior:
        - Vector fields use frozen effective input.
        - Trajectory overlays are shown only during time points whose effective
          input matches that panel.
        - Autonomous overlays exclude nonzero-input and reward-feedback epochs.
        - Fixed points are computed separately for each frozen-input panel.
    """
    task = task.lower()

    if model.m.shape[1] != 2:
        raise ValueError("This plotter assumes rank == 2.")

    model.eval()

    device = _model_device(model)
    dtype = _model_dtype(model)
    input_size = inputs.shape[-1]

    # ------------------------------------------------------------------
    # Run model and project trajectories.
    # ------------------------------------------------------------------
    with torch.no_grad():
        readouts, rates, rec_inputs, effective_x = run_low_rank_with_effective_inputs(
            model,
            inputs,
            targets=targets,
        )

    kappa_traj = (
        project_rec_inputs_to_kappa(model, rec_inputs)
        .detach()
        .cpu()
        .numpy()
    )

    # ------------------------------------------------------------------
    # Axis limits.
    # ------------------------------------------------------------------
    if xlim is None or ylim is None:
        auto_xlim, auto_ylim = auto_limits_from_trajs(
            kappa_traj[..., :2],
            qlo=qlo,
            qhi=qhi,
            pad=pad,
            symmetric=symmetric_auto,
        )

        xlim = auto_xlim if xlim is None else xlim
        ylim = auto_ylim if ylim is None else ylim

    # ------------------------------------------------------------------
    # Conditions and panel specs.
    # ------------------------------------------------------------------
    cond_idx, meta = condition_indices_for_task(
        inputs.detach().cpu(),
        timing=timing,
        task=task,
        condition_names=condition_names,
        dual_mode=dual_mode,
    )

    specs = flow_specs_for_task(
        timing=timing,
        task=task,
        input_size=input_size,
        cond_idx=cond_idx,
        meta=meta,
        dual_mode=dual_mode,
    )

    max_dim = max(
        [d for spec in specs if spec["dims"] is not None for d in spec["dims"]],
        default=-1,
    )

    if input_size <= max_dim:
        raise ValueError(
            f"Task {task} requires input_size > {max_dim}, got {input_size}."
        )

    colors = _condition_colors(cond_idx.keys())

    # ------------------------------------------------------------------
    # Precompute fields, fixed points, masks.
    # ------------------------------------------------------------------
    caches = []
    all_speeds = []

    for spec in specs:
        ff_input = make_input(
            input_size,
            active_dims=spec["dims"],
            value=1.0,
            device=device,
            dtype=dtype,
        )

        K1, K2, U, V, speed, params, ff_input_np = make_vector_field_grid(
            model,
            ff_input=ff_input,
            xlim=xlim,
            ylim=ylim,
            n_grid=n_grid,
            include_beta=include_beta_in_field,
        )

        fixed_points, fixed_residuals = find_all_fixed_points(
            model,
            xlim=xlim,
            ylim=ylim,
            ff_input=ff_input,
            n_seeds=n_fp_seeds,
            residual_tol=1e-8,
            merge_tol=5e-2,
        )

        fp_labels, fp_eigvals = classify_fixed_points(
            model,
            fixed_points,
            ff_input=ff_input,
            eig_tol=1e-5,
        )

        panel_mask = _canonical_input_mask(
            effective_x,
            dims=spec["dims"],
            threshold=input_threshold,
            atol_inactive=inactive_atol,
        )

        caches.append(
            dict(
                spec=spec,
                ff_input=ff_input,
                K1=K1,
                K2=K2,
                U=U,
                V=V,
                speed=speed,
                fixed_points=fixed_points,
                fixed_residuals=fixed_residuals,
                fp_labels=fp_labels,
                fp_eigvals=fp_eigvals,
                panel_mask=panel_mask.detach().cpu().numpy(),
            )
        )

        all_speeds.append(speed[np.isfinite(speed)].reshape(-1))

    speed_vmax = np.percentile(np.concatenate(all_speeds), speed_percentile)

    # ------------------------------------------------------------------
    # Figure layout.
    # ------------------------------------------------------------------
    n_panels = len(specs)

    fig = plt.figure(
        figsize=(n_panels * figsize_per_panel + 0.45, figsize_per_panel),
        constrained_layout=False,
    )
    fig.set_layout_engine("none")

    gs = fig.add_gridspec(
        1,
        n_panels + 1,
        width_ratios=[1.0] * n_panels + [0.045],
        left=0.045,
        right=0.965,
        bottom=0.16,
        top=0.84,
        wspace=0.16,
    )

    axes = []
    for i in range(n_panels):
        if i == 0:
            ax_i = fig.add_subplot(gs[0, i])
        else:
            ax_i = fig.add_subplot(gs[0, i], sharex=axes[0], sharey=axes[0])
        axes.append(ax_i)

    axes = np.asarray(axes, dtype=object)
    cax = fig.add_subplot(gs[0, -1])

    last_hm = None

    # Draw panels.
    # ------------------------------------------------------------------
    for ax, cache in zip(axes, caches):

        K2 = cache["K2"]

            shading="auto",
            cmap="magma",

        )

        k1 = K1[0, :]
        k2 = K2[:, 0]

        ax.streamplot(
            k1,
            k2,
            U,
            V,
            color="white",
            density=1.05,
            linewidth=0.70,
            arrowsize=0.80,
            zorder=2,
        )

        # --------------------------------------------------------------
        # Fixed points.
        # --------------------------------------------------------------
        fixed_points = cache["fixed_points"]
        fp_labels = cache["fp_labels"]

        for label, marker in [
            ("attractor", "o"),
            ("saddle", "x"),
            ("repeller", "^"),

        ]:
            if len(fp_labels) == 0:
                continue

            mask = fp_labels == label
            if not np.any(mask):
                continue

            pts = fixed_points[mask]

                    edgecolors="cyan",
                    linewidths=1.1,

        # --------------------------------------------------------------
        conds = list(spec["conds"])

        if spec["name"] == "Autonomous" and max_autonomous_conditions is not None:
            conds = conds[:max_autonomous_conditions]

                        ax.plot(
                            seg_i[:, 0],

@torch.no_grad()
def rank2_kappa_flow_sliced(
    model,
    x_t,
    q_slice="same",
    kappa1_lim=(-3.0, 3.0),
    kappa2_lim=(-3.0, 3.0),
    grid_size=31,
    device=None,
):
    """
    Rank-2 projected kappa flow for the augmented dynamics.

    State variables:
        q     : recurrent-input coordinate, h = m q
        kappa : rate overlap, kappa = n^T r / N

    Dynamics:
        q_next = beta_h q + (1 - beta_h) kappa
        kappa_next = beta_r kappa + (1 - beta_r) F(q_next)

    q_slice:
        "same" -> q = kappa
        "zero" -> q = 0
        Tensor of shape [2] or [G, 2] -> custom slice
    """
    assert model.rank == 2

    if device is None:
        device = model.device

    N = model.hidden_size
    m = model.m.detach()
    n = model.n.detach()

    if x_t.ndim == 1:
        x_t = x_t[None, :]
    x_t = x_t.to(device)

    k1 = torch.linspace(kappa1_lim[0], kappa1_lim[1], grid_size, device=device)

# ============================================================
# Small utility: temporarily disable noise / reward feedback

@contextmanager
def temporary_attrs(obj, **kwargs):
    old = {k: getattr(obj, k) for k in kwargs}
    try:
        for k, v in kwargs.items():
            setattr(obj, k, v)
        yield
    finally:
        for k, v in old.items():
            setattr(obj, k, v)

@torch.no_grad()
def rank2_kappa_delta_torch(model, x_t, kappa, q_slice="same"):
    """
    Compute Delta kappa for the rank-2 projected dynamics.

    kappa : [G, 2]

    q_slice:
        "same" -> q = kappa, standard low-rank/slaved closure
        "zero" -> q = 0
    """
    assert model.rank == 2, "This helper assumes model.rank == 2."

    device = model.device
    kappa = kappa.to(device)

    if x_t.ndim == 1:
        x_t = x_t[None, :]
    x_t = x_t.to(device)

    G = kappa.shape[0]
    N = model.hidden_size

    m = model.m.detach()  # [N, 2]
    n = model.n.detach()  # [N, 2]

    if q_slice == "same":
        q = kappa
    elif q_slice == "zero":
        q = torch.zeros_like(kappa)
    else:
        raise ValueError("q_slice must be 'same' or 'zero'.")

    beta_h = model.exp_alpha_rec.to(device)
    beta_r = model.exp_alpha.to(device)

    # recurrent-input coordinate update
    q_next = beta_h * q + (1.0 - beta_h) * kappa

    # input drive
    if model.wi is not None:
        x_grid = x_t.expand(G, -1)
        input_drive = model.Ai * model.wi(x_grid)  # [G, N]
    else:
        input_drive = 0.0

    # h = m q_next
    h_next = q_next @ m.T  # [G, N]

    # phi = tanh(I + gain h)
    phi = torch.tanh(input_drive + model.gain * h_next)

    # F(kappa) = n^T phi / N
    F = phi @ n / N  # [G, 2]

    # rate-overlap update
    kappa_next = beta_r * kappa + (1.0 - beta_r) * F

    return kappa_next - kappa


@torch.no_grad()
def rank2_kappa_flow(
    model,
    x_t,
    kappa1_lim=(-3.0, 3.0),
    kappa2_lim=(-3.0, 3.0),
    grid_size=61,
    q_slice="same",
):
    """
    Compute rank-2 flow field in the (kappa_1, kappa_2) plane.

    Returns:
        K1, K2, U, V, speed
    """
    assert model.rank == 2, "This helper assumes model.rank == 2."

    device = model.device

    k1 = torch.linspace(kappa1_lim[0], kappa1_lim[1], grid_size, device=device)
    k2 = torch.linspace(kappa2_lim[0], kappa2_lim[1], grid_size, device=device)

    K1, K2 = torch.meshgrid(k1, k2, indexing="ij")

    kappa = torch.stack(
        [K1.reshape(-1), K2.reshape(-1)],
        dim=-1,
    )  # [G, 2]

    d_kappa = rank2_kappa_delta_torch(
        model,
        x_t=x_t,
        kappa=kappa,
        q_slice=q_slice,
    )

    U = d_kappa[:, 0].reshape(grid_size, grid_size)
    V = d_kappa[:, 1].reshape(grid_size, grid_size)
    speed = torch.sqrt(U**2 + V**2)

    return (
        K1.cpu().numpy(),
        K2.cpu().numpy(),
        U.cpu().numpy(),
        V.cpu().numpy(),
        speed.cpu().numpy(),
    )

def classify_fixed_point(model, x_t, kappa_star, q_slice="same", eps=1e-4):
    """
    Classify fixed point using finite-difference Jacobian of the discrete map:

        kappa_{t+1} = kappa_t + Delta kappa(kappa_t)

    Stability criterion:
        stable if all eigenvalues of J_map have abs < 1.
    """
    kappa_star = np.asarray(kappa_star, dtype=np.float64)

    J_delta = np.zeros((2, 2), dtype=np.float64)

    for j in range(2):
        step = np.zeros(2)
        step[j] = eps

        kp = torch.tensor(
            kappa_star + step,
            dtype=torch.float32,
            device=model.device,
        )[None, :]

        km = torch.tensor(
            kappa_star - step,
            dtype=torch.float32,
            device=model.device,
        )[None, :]

        fp = rank2_kappa_delta_torch(
            model,
            x_t,
            kp,
            q_slice=q_slice,
        ).cpu().numpy()[0]

        fm = rank2_kappa_delta_torch(
            model,
            x_t,
            km,
            q_slice=q_slice,
        ).cpu().numpy()[0]

        J_delta[:, j] = (fp - fm) / (2.0 * eps)

    J_map = np.eye(2) + J_delta
    eigvals = np.linalg.eigvals(J_map)
    abs_eigs = np.abs(eigvals)

    if np.all(abs_eigs < 1.0):
        kind = "stable"
    elif np.all(abs_eigs > 1.0):
        kind = "unstable"
    else:
        kind = "saddle"

    return eigvals, kind


def find_rank2_fixed_points(
    model,
    x_t,
    kappa1_lim=(-3, 3),
    kappa2_lim=(-3, 3),
    n_init=21,
    q_slice="same",
    root_tol=1e-8,
    residual_tol=1e-5,
    duplicate_tol=1e-3,
    pad=0.25,
):
    """
    Find fixed points of the projected rank-2 kappa field:

        Delta kappa(kappa; x_t) = 0
    """
    assert model.rank == 2, "This helper assumes model.rank == 2."

    def f_np(z):
        z_t = torch.tensor(
            z,
            dtype=torch.float32,
            device=model.device,
        )[None, :]

        dz = rank2_kappa_delta_torch(
            model,
            x_t=x_t,
            kappa=z_t,
            q_slice=q_slice,
        )

        return dz.cpu().numpy()[0].astype(np.float64)

    k1s = np.linspace(kappa1_lim[0], kappa1_lim[1], n_init)
    k2s = np.linspace(kappa2_lim[0], kappa2_lim[1], n_init)

    fixed_points = []

    for k1 in k1s:
        for k2 in k2s:
            z0 = np.array([k1, k2], dtype=np.float64)

            sol = root(f_np, z0, method="hybr", tol=root_tol)

            if not sol.success:
                continue

            z = sol.x.astype(np.float64)
            resid = float(np.linalg.norm(f_np(z)))

            in_bounds = (
                (kappa1_lim[0] - pad <= z[0] <= kappa1_lim[1] + pad)
                and
                (kappa2_lim[0] - pad <= z[1] <= kappa2_lim[1] + pad)
            )

            if resid > residual_tol or not in_bounds:
                continue

            # Deduplicate nearby roots.
            is_new = True
            for fp in fixed_points:
                if np.linalg.norm(z - fp["kappa"]) < duplicate_tol:
                    is_new = False
                    if resid < fp["residual"]:
                        fp["kappa"] = z
                        fp["residual"] = resid
                    break

            if is_new:
                eigvals, kind = classify_fixed_point(
                    model,
                    x_t=x_t,
                    kappa_star=z,
                    q_slice=q_slice,
                )

                fixed_points.append(
                    {
                        "kappa": z,
                        "residual": resid,
                        "eigvals": eigvals,
                        "kind": kind,
                    }
                )

    fixed_points = sorted(
        fixed_points,
        key=lambda d: (d["kappa"][0], d["kappa"][1]),
    )

    return fixed_points

@torch.no_grad()
def get_kappa_trajectories(model, ff_inputs, targets=None, disable_rwd=False):
    """
    Run the real model and project rates to:

        kappa_t = r_t @ n / N

    Returns:
        kappa: np.ndarray [B, T, 2]
        rates, rec_inputs, readout
    """
    ff_inputs = ff_inputs.to(model.device)

    if targets is not None:
        targets = targets.to(model.device)

    attrs = {"noise": 0.0}
    if disable_rwd:
        attrs["rwd"] = False

    with temporary_attrs(model, **attrs):
        readout, rates, rec_inputs = model(
            ff_inputs,
            targets=targets,
            ret_rates=True,
        )

    kappa = rates @ model.n.detach() / model.hidden_size

    return kappa.cpu().numpy(), rates, rec_inputs, readout


def infer_dpa_labels_from_inputs(
    ff_inputs,
    timing=None,
    sample_channels=(0, 1),
    test_channels=(2, 3),
):
    """
    Infer AC/AD/BC/BD labels from DPA input channels.

    Assumes:
        channel 0 = A
        channel 1 = B
        channel 2 = C
        channel 3 = D
    """
    if torch.is_tensor(ff_inputs):
        x = ff_inputs.detach().cpu()
    else:
        x = torch.as_tensor(ff_inputs)

    if timing is not None:
        n_on = timing.n_stim_on
        n_off = timing.n_stim_off

        sample_slice = slice(int(n_on[0]), int(n_off[0]))

        # DPA task: second stim is test.
        # Dual task: fourth stim is test.
        if len(n_on) >= 4:
            test_slice = slice(int(n_on[3]), int(n_off[3]))
        else:
            test_slice = slice(int(n_on[1]), int(n_off[1]))

        sample_score = x[:, sample_slice, list(sample_channels)].mean(dim=1)
        test_score = x[:, test_slice, list(test_channels)].mean(dim=1)
    else:
        sample_score = x[:, :, list(sample_channels)].mean(dim=1)
        test_score = x[:, :, list(test_channels)].mean(dim=1)

    sample_idx = sample_score.argmax(dim=1).numpy()
    test_idx = test_score.argmax(dim=1).numpy()

    sample_char = np.where(sample_idx == 0, "A", "B")
    test_char = np.where(test_idx == 0, "C", "D")

    return np.char.add(sample_char, test_char)


def dpa_labels_from_condition_names(condition_names):
    """
    Convert names like:
        A_C
        A_go_C
        B_nogo_D

    into:
        AC, AD, BC, BD
    """
    labels = []

    for name in condition_names:
        s = str(name)

        if s.startswith("A"):
            sample = "A"
        elif s.startswith("B"):
            sample = "B"
        else:
            raise ValueError(f"Could not parse sample from condition name: {s}")

        if s.endswith("C"):
            test = "C"
        elif s.endswith("D"):
            test = "D"
        else:
            raise ValueError(f"Could not parse test from condition name: {s}")

        labels.append(sample + test)

    return np.asarray(labels)

    overlay_individual=False,
    trial_inds=None,
    avg_trial_types=("AC", "AD", "BC", "BD"),

        x_t=x_input,
        kappa1_lim=kappa1_lim,
        kappa2_lim=kappa2_lim,
        grid_size=grid_size,
        q_slice=q_slice,
    )

    # Choose heatmap

    elif heatmap_kind == "dK1":
        Ha, Hi = Ua, Ui
        cbar_label = r"$\Delta \kappa_1$"
    elif heatmap_kind == "dK2":
        Ha, Hi = Va, Vi
        cbar_label = r"$\Delta \kappa_2$"

        raise ValueError("heatmap_kind must be 'speed', 'dK1', or 'dK2'.")

        vmin = min(np.nanmin(Ha), np.nanmin(Hi))
        vmax = max(np.nanmax(Ha), np.nanmax(Hi))
    else:
        vmin = vmax = None

    # -------------------------
    # Plot
    # -------------------------

        1,
        2,
        figsize=(11.5, 5),
        constrained_layout=True,
        sharex=True,
        sharey=True,

        axs[0],
        K1a,
        K2a,
        Ua,
        Va,
        heatmap=Ha,
        title="Autonomous / baseline",
        cmap=cmap,
        flow_color=flow_color,
        normalize_flow=normalize_flow,
        vmin=vmin,

    # Trajectory overlays
    # -------------------------
    if ff_inputs is not None:
        kappa, rates, rec_inputs, readout = get_kappa_trajectories(
            model,
            ff_inputs=ff_inputs,
            targets=targets,
            disable_rwd=True,
        )

        else:
            dpa_labels = infer_dpa_labels_from_inputs(
                ff_inputs,

import torch


@torch.no_grad()
def init_dpa_internal_readout_prepost(
    model,
    mem: int = 0,
    out: int = 1,
    memory_lambda: float = 0.990,
    decision_lambda: float = 0.900,
    target_mn_corr: float = 0.5,
    target_out_mn_corr: float | None = 0.5,
    sample_scale: float = 0.8,
    test_scale: float = 0.8,
    readout_scale: float = 1.0,
    mix_strength: float = 0.10,
    readout_private_scale: float = 0.02,
    noise_scale_mn: float = 0.05,
    noise_scale_in: float = 0.05,
    seed: int | None = None,
    verbose: bool = True,
):
    """
    DPA initializer for internal-readout low-rank RNNs.

    Assumes input channels:
        0 = A
        1 = B
        2 = C
        3 = D

    Rank roles:
        mem:
            sample-memory rank.

        out:
            behavioral/internal readout rank.

    Important:
        target_mn_corr controls corr(m_mem, n_mem).
        target_out_mn_corr controls corr(m_out, n_out).

    The recurrent overlap matrix is

        C = n.T @ m / N

    and this initializer sets approximately

        C[mem, mem] = memory_lambda
        C[out, out] = decision_lambda
        C[mem, out] = C[out, mem] = 0
    """
    assert model.rank >= 2, "Need rank >= 2."
    assert model.input_size >= 4, "Need input channels A,B,C,D."
    assert mem != out, "mem and out ranks must be different."
    assert 0.0 < target_mn_corr <= 1.0
    assert target_out_mn_corr is None or (0.0 < target_out_mn_corr <= 1.0)
    assert 0.0 <= mix_strength <= 1.0
    assert memory_lambda > 0.0
    assert decision_lambda >= 0.0
    assert readout_scale > 0.0

    device = model.device
    N = model.hidden_size

    gen = None
    if seed is not None:
        gen = torch.Generator(device=device)
        gen.manual_seed(seed)

    def randn(*shape):
        return torch.randn(*shape, device=device, generator=gen)

    def zscore(v):
        v = v - v.mean()
        return v / v.std().clamp_min(1e-6)

    def orthogonalize(v, basis):
        for b in basis:
            denom = (b @ b).clamp_min(1e-6)
            v = v - (v @ b) / denom * b
        return v

    def corr(x, y):
        x = x.detach().flatten().cpu()
        y = y.detach().flatten().cpu()
        x = x - x.mean()
        y = y - y.mean()
        return (x @ y / (x.norm() * y.norm()).clamp_min(1e-8)).item()

    # ------------------------------------------------------------------
    # 1. Construct sample-memory rank with controlled corr(m_mem, n_mem).
    # ------------------------------------------------------------------
    u_mem = zscore(randn(N))

    p_m = randn(N)
    p_m = orthogonalize(p_m, [u_mem])
    p_m = zscore(p_m)

    p_n = randn(N)
    p_n = orthogonalize(p_n, [u_mem, p_m])
    p_n = zscore(p_n)

    # If u_mem, p_m, p_n are orthogonal unit-variance vectors:
    #
    #   m = a u + s p_m
    #   n = a u + s p_n
    #
    # then
    #
    #   n.T m / N ≈ a^2
    #   corr(m,n) ≈ a^2 / (a^2 + s^2)
    a = memory_lambda ** 0.5
    s = (memory_lambda * (1.0 / target_mn_corr - 1.0)) ** 0.5

    m_mem = a * u_mem + s * p_m
    n_mem = a * u_mem + s * p_n

    # Correct finite-N deviation so C[mem,mem] is exact.
    C_mem = (n_mem @ m_mem) / N
    n_mem = n_mem * (memory_lambda / C_mem.clamp_min(1e-8))

    # ------------------------------------------------------------------
    # 2. Test direction, orthogonal to memory components.
    # ------------------------------------------------------------------
    u_test = randn(N)
    u_test = orthogonalize(u_test, [u_mem, p_m, p_n])
    u_test = zscore(u_test)

    # Optional weak product direction for DPA product readout.
    u_mix = u_mem * u_test
    u_mix = orthogonalize(u_mix, [u_mem, p_m, p_n, u_test])
    u_mix = zscore(u_mix)

    # Private direction for output rank.
    u_out_private = randn(N)
    u_out_private = orthogonalize(
        u_out_private,
        [u_mem, p_m, p_n, u_test, u_mix],
    )
    u_out_private = zscore(u_out_private)

    # Random readout direction orthogonal to structured directions.
    u_noise = randn(N)
    u_noise = orthogonalize(
        u_noise,
        [u_mem, p_m, p_n, u_test, u_mix, u_out_private],
    )
    u_noise = zscore(u_noise)

    # Optional weak product bias in the internal readout.
    u_readout = (
        mix_strength * u_mix
        + (1.0 - mix_strength**2) ** 0.5 * u_noise
    )
    u_readout = zscore(u_readout)

    # ------------------------------------------------------------------
    # 3. Reset all model parameters weakly.
    # ------------------------------------------------------------------
    model.m.normal_(0.0, noise_scale_mn)
    model.n.normal_(0.0, noise_scale_mn)

    if model.wi is not None:
        model.wi.weight.normal_(0.0, noise_scale_in)
        model.wi.bias.zero_()

    if model.wo is not None:
        model.wo.weight.normal_(0.0, noise_scale_in)
        model.wo.bias.zero_()

    # ------------------------------------------------------------------
    # 4. Install sample-memory rank.
    # ------------------------------------------------------------------
    model.m[:, mem] = m_mem
    model.n[:, mem] = n_mem

    # ------------------------------------------------------------------
    # 5. Install decision/readout rank with controlled corr(m_out,n_out).
    #
    # Readout:
    #
    #     z_out = n_out.T r / N
    #
    # Recurrent self-overlap:
    #
    #     C[out,out] = n_out.T m_out / N = decision_lambda
    #
    # We choose
    #
    #     m_out = q_out + private_scale * p_out
    #
    # where q_out is parallel to n_out and p_out is orthogonal to n_out.
    # ------------------------------------------------------------------
    n_out = readout_scale * u_readout

    # Private direction orthogonal to n_out.
    p_out = u_out_private.clone()
    p_out = orthogonalize(p_out, [n_out])
    p_out = zscore(p_out)

    n_var = (n_out @ n_out) / N

    # Parallel component required for desired C[out,out].
    q_out = (decision_lambda / n_var.clamp_min(1e-8)) * n_out

    if target_out_mn_corr is not None:
        rho = float(target_out_mn_corr)

        # With q_out ⟂ p_out and p_out unit variance:
        #
        #   corr(q_out + s p_out, n_out)
        #       ≈ sqrt(var(q_out) / (var(q_out) + s^2))
        #
        # Solve for s.
        q_var = (q_out @ q_out) / N
        private_scale = torch.sqrt(
            q_var * (1.0 / (rho**2) - 1.0)
        )
    else:
        private_scale = torch.as_tensor(
            readout_private_scale,
            device=device,
            dtype=q_out.dtype,
        )

    m_out = q_out + private_scale * p_out

    # Correct finite-N deviation so C[out,out] is exact.
    C_out = (n_out @ m_out) / N
    m_out = m_out + ((decision_lambda - C_out) / n_var.clamp_min(1e-8)) * n_out

    model.n[:, out] = n_out
    model.m[:, out] = m_out

    # ------------------------------------------------------------------
    # 6. Install DPA inputs.
    #
    # Pairing:
    #   A,C -> + * + = +
    #   B,D -> - * - = +
    #   A,D -> + * - = -
    #   B,C -> - * + = -
    # ------------------------------------------------------------------
    model.wi.weight[:, 0] = +sample_scale * u_mem   # A
    model.wi.weight[:, 1] = -sample_scale * u_mem   # B
    model.wi.weight[:, 2] = +test_scale   * u_test  # C
    model.wi.weight[:, 3] = -test_scale   * u_test  # D

    # Other channels, e.g. Go/NoGo/Cue/reward, stay weak random.
    if model.input_size > 4:
        model.wi.weight[:, 4:].normal_(0.0, noise_scale_in)

    # ------------------------------------------------------------------
    # 7. Diagnostics.
    # ------------------------------------------------------------------
    C = (model.n.T @ model.m) / model.hidden_size
    eig_C = torch.linalg.eigvals(C)

    diagnostics = {
        "C": C.detach().clone(),
        "eig_C": eig_C.detach().clone(),

        "C_mem_mem": C[mem, mem].item(),
        "C_out_out": C[out, out].item(),
        "C_mem_out": C[mem, out].item(),
        "C_out_mem": C[out, mem].item(),

        "corr_m_mem_n_mem": corr(model.m[:, mem], model.n[:, mem]),
        "corr_m_out_n_out": corr(model.m[:, out], model.n[:, out]),

        "corr_m_mem_n_out": corr(model.m[:, mem], model.n[:, out]),
        "corr_m_out_n_mem": corr(model.m[:, out], model.n[:, mem]),

        "corr_inA_n_out": corr(model.wi.weight[:, 0], model.n[:, out]),
        "corr_inB_n_out": corr(model.wi.weight[:, 1], model.n[:, out]),
        "corr_inC_n_out": corr(model.wi.weight[:, 2], model.n[:, out]),
        "corr_inD_n_out": corr(model.wi.weight[:, 3], model.n[:, out]),

        "corr_inA_m_mem": corr(model.wi.weight[:, 0], model.m[:, mem]),
        "corr_inB_m_mem": corr(model.wi.weight[:, 1], model.m[:, mem]),
        "corr_inC_u_test": corr(model.wi.weight[:, 2], u_test),
        "corr_inD_u_test": corr(model.wi.weight[:, 3], u_test),
    }

    if verbose:
        print("Low-rank recurrent core C = n^T m / N:")
        print(C.detach().cpu())
        print("eig(C):", eig_C.detach().cpu())
        print()
        print(f"C[{mem},{mem}] = {C[mem, mem].item():+.4f}  # sample memory")
        print(f"C[{out},{out}] = {C[out, out].item():+.4f}  # decision memory")
        print(f"C[{mem},{out}] = {C[mem, out].item():+.4f}")
        print(f"C[{out},{mem}] = {C[out, mem].item():+.4f}")
        print()
        print(f"corr(m{mem}, n{mem}) = {diagnostics['corr_m_mem_n_mem']:+.4f}")
        print(f"corr(m{out}, n{out}) = {diagnostics['corr_m_out_n_out']:+.4f}")
        print(f"corr(m{mem}, n{out}) = {diagnostics['corr_m_mem_n_out']:+.4f}")
        print(f"corr(m{out}, n{mem}) = {diagnostics['corr_m_out_n_mem']:+.4f}")
        print()
        print(f"corr(inA, n{out}) = {diagnostics['corr_inA_n_out']:+.4f}")
        print(f"corr(inB, n{out}) = {diagnostics['corr_inB_n_out']:+.4f}")
        print(f"corr(inC, n{out}) = {diagnostics['corr_inC_n_out']:+.4f}")
        print(f"corr(inD, n{out}) = {diagnostics['corr_inD_n_out']:+.4f}")

    return {
        "u_mem": u_mem,
        "p_m": p_m,
        "p_n": p_n,
        "m_mem": m_mem,
        "n_mem": n_mem,
        "u_test": u_test,
        "u_mix": u_mix,
        "u_out_private": u_out_private,
        "u_noise": u_noise,
        "u_readout": u_readout,
        "n_out": n_out,
        "m_out": m_out,
        "diagnostics": diagnostics,
    }

def concrete_conditions(pair_status: str, gng_status: str) -> list[str]:
    """
    Return the two concrete condition names for a panel.

    pair_status: "pair" or "unpair"
    gng_status : "none", "go", or "nogo"
    """
    if pair_status == "pair":
        base = [("A", "C"), ("B", "D")]
    elif pair_status == "unpair":
        base = [("A", "D"), ("B", "C")]
    else:
        raise ValueError(pair_status)

    out = []
    for sample, test in base:
        if gng_status == "none":
            out.append(f"{sample}_{test}")
        else:
            out.append(f"{sample}_{gng_status}_{test}")

    return out

def plot_dual_kappa_targets(
    kappa,
    targets,
    condition_names,
    x_time,
    dim: int = 0,
    kappa_label: str = r"\kappa",
    T_STIM_ON=None,
    T_STIM_OFF=None,
    width: float = 6,
    height: float | None = None,
    condition_colors=("tab:blue", "tab:orange"),
    condition_styles=("-", "-"),
    qlim: float = 99,
):
    """
    Plot dual-task condition-averaged latent coordinate and targets.

    Layout:
        columns = DPA-only, Go, NoGo
        rows    = paired (AC, BD), unpaired (AD, BC)

    Targets and latent coordinate are plotted on the same axis.
    """
    if height is None:
        golden_ratio = (5**0.5 - 1) / 2
        height = width * golden_ratio

    if torch.is_tensor(kappa):
        kappa_np = kappa.detach().cpu().numpy()
    else:
        kappa_np = np.asarray(kappa)

    if torch.is_tensor(targets):
        targets_np = targets.detach().cpu().numpy()
    else:
        targets_np = np.asarray(targets)

    condition_names = np.asarray(condition_names).astype(str)
    x_time = np.asarray(x_time)

    if kappa_np.ndim != 3:
        raise ValueError(
            f"kappa must have shape [trials, time, rank], got {kappa_np.shape}"
        )
    if dim >= kappa_np.shape[-1] or dim < -kappa_np.shape[-1]:
        raise ValueError(f"dim={dim} outside available rank={kappa_np.shape[-1]}")

    def concrete_conditions(pair_status: str, gng_status: str) -> list[str]:
        if pair_status == "pair":
            base = [("A", "C"), ("B", "D")]
        elif pair_status == "unpair":
            base = [("A", "D"), ("B", "C")]
        else:
            raise ValueError(pair_status)

        if gng_status == "none":
            return [f"{sample}_{test}" for sample, test in base]
        return [f"{sample}_{gng_status}_{test}" for sample, test in base]

    row_specs = {
        "Pair": "pair",
        "Unpair": "unpair",
    }

    col_specs = {
        "DPA only": "none",
        "Go": "go",
        "NoGo": "nogo",
    }

    fig, ax = plt.subplots(
        2,
        3,
        figsize=(3 * width, 2 * height),
        sharex=True,
        sharey=True,
    )

    for row, (row_title, pair_status) in enumerate(row_specs.items()):
        for col, (col_title, gng_status) in enumerate(col_specs.items()):
            this_ax = ax[row, col]
            conds = concrete_conditions(pair_status, gng_status)

            for cond_i, cond_name in enumerate(conds):
                idxs = np.where(condition_names == cond_name)[0]

                if len(idxs) == 0:
                    this_ax.text(
                        0.5,
                        0.5 - 0.12 * cond_i,
                        f"Missing {cond_name}",
                        ha="center",
                        va="center",
                        transform=this_ax.transAxes,
                        color="crimson",
                    )
                    continue

                k_mean = kappa_np[idxs, :, dim].mean(axis=0)

                # Targets often have only one channel; use dim when possible,
                # otherwise fall back to the last target channel.
                target_dim = dim if -targets_np.shape[-1] <= dim < targets_np.shape[-1] else -1
                tgt_mean = targets_np[idxs, :, target_dim].mean(axis=0)

                ls = condition_styles[cond_i]
                color = condition_colors[cond_i]

                # Target.
                this_ax.plot(
                    x_time,
                    tgt_mean,
                    color="k",
                    ls=ls,
                    lw=2.4,
                    alpha=0.85,
                )

                # Latent coordinate / internal readout.
                this_ax.plot(
                    x_time,
                    k_mean,
                    color=color,
                    ls=ls,
                    lw=3.0,
                    alpha=0.9,
                )

            if T_STIM_ON is not None and T_STIM_OFF is not None:
                for t_on, t_off in zip(T_STIM_ON, T_STIM_OFF):
                    this_ax.axvspan(
                        float(t_on),
                        float(t_off),
                        alpha=0.12,
                        color="tab:blue",
                        lw=0,
                    )

            this_ax.set_title(col_title)

            if col == 0:
                this_ax.set_ylabel(
                    rf"Target, ${kappa_label}_{dim + 1}$" + f"\n{row_title} trials"
                )

            if row == 1:
                this_ax.set_xlabel("Time (s)")

    # Common y-limits on the shared single axis.
    k_vals = kappa_np[:, :, dim].reshape(-1)
    k_lim = np.nanpercentile(np.abs(k_vals), qlim)
    y_lim = 1.1 * max(float(k_lim), 1.5, 1e-6)

    for this_ax in ax.flat:
        this_ax.set_ylim([-y_lim, y_lim])

    legend_handles = [
        Line2D([0], [0], color="k", lw=2.4, label="Target"),
        Line2D([0], [0], color=condition_colors[0], lw=3.0, label="A"),
        Line2D([0], [0], color=condition_colors[1], lw=3.0, label="B"),
    ]

    fig.legend(
        handles=legend_handles,
        loc="upper center",
        ncol=6,
        frameon=False,
        fontsize=10,
        bbox_to_anchor=(0.5, 1.04),
    )

    plt.tight_layout()
    return fig, ax
#+end_src

@torch.no_grad()
def eval_dpa_accuracy(
    model,
    timing,
    n_trials=1024,
    input_size=7,
    noise=0.0,
    readout_index=1,
    device="cpu",
):
    model.eval()

    X, y = generate_dpa_trials(
        n_trials,
        timing=timing,
        input_size=input_size,
        noise=noise,
        target_rank=1,
    )

    out = model(X.to(device), y.to(device))
    pred = out[..., readout_index].detach().cpu()

    decision_start = int(timing.n_stim_off[1])
    pred_final = pred[:, decision_start:].mean(dim=1)
    target_final = y[:, -1, -1]

    acc = ((pred_final > 0) == (target_final > 0)).float().mean().item()

    return {
        "acc": acc,
        "pred_final": pred_final,
        "target_final": target_final,
    }


@torch.no_grad()
def eval_gng_accuracy(
    model,
    timing,
    n_trials=1024,
    input_size=7,
    noise=0.0,
    readout_index=1,
    device="cpu",
):
    model.eval()

    X, y = generate_gng_trials(
        n_trials,
        timing=timing,
        input_size=input_size,
        noise=noise,
        target_rank=1,
    )

    out = model(X.to(device), y.to(device))
    pred = out[..., readout_index].detach().cpu()

    # Go/NoGo identity from inputs.
    stim_epoch = slice(int(timing.n_stim_on[0]), int(timing.n_stim_off[0]))
    stim_drive = X[:, stim_epoch, 4:6].mean(dim=1)
    is_go = stim_drive[:, 0] > stim_drive[:, 1]
    is_nogo = ~is_go

    # Evaluate after cue offset.
    decision_start = int(timing.n_stim_off[1])
    pred_final = pred[:, decision_start:].mean(dim=1)

    # Go should be high, NoGo should be near 0 or below threshold.
    pred_label = pred_final > 0.5
    acc = (pred_label == is_go).float().mean().item()

    return {
        "acc": acc,
        "pred_final": pred_final,
        "is_go": is_go,
    }


@torch.no_grad()
def eval_dual_accuracy(
    model,
    timing,
    n_trials=1024,
    input_size=7,
    noise=0.0,
    readout_index=1,
    device="cpu",
):
    model.eval()

    X, y, trial_type, condition_names = generate_dual_trials(
        n_trials,
        timing=timing,
        input_size=input_size,
        noise=noise,
        target_rank=1,
    )

    out = model(X.to(device), y.to(device))
    pred = out[..., readout_index].detach().cpu()

    names = np.asarray(condition_names).astype(str)

    # DPA final epoch: after test offset.
    dpa_start = int(timing.n_stim_off[3])
    pred_dpa = pred[:, dpa_start:].mean(dim=1)
    target_dpa = y[:, -1, -1]
    dpa_acc = ((pred_dpa > 0) == (target_dpa > 0)).float().mean().item()

    # GNG reward epoch: between cue offset and DPA test onset.
    rwd_start = int(timing.n_stim_off[2])
    rwd_stop = int(timing.n_stim_on[3])
    pred_gng = pred[:, rwd_start:rwd_stop].mean(dim=1)

    is_go = torch.as_tensor(["_go_" in n for n in names])
    is_nogo = torch.as_tensor(["_nogo_" in n for n in names])
    is_gng = is_go | is_nogo

    if is_gng.any():
        gng_pred_label = pred_gng[is_gng] > 0.5
        gng_true_label = is_go[is_gng]
        gng_acc = (gng_pred_label == gng_true_label).float().mean().item()
    else:
        gng_acc = float("nan")

    return {
        "dpa_acc": dpa_acc,
        "gng_acc": gng_acc,
        "pred_dpa": pred_dpa,
        "target_dpa": target_dpa,
        "pred_gng": pred_gng,
        "condition_names": condition_names,
    }

def load_checkpoint_for_seed(model, path, seed, filename_template, device):
    ckpt_path = os.path.join(path, filename_template.format(seed=seed))
    if not os.path.exists(ckpt_path):
        raise FileNotFoundError(f"Missing checkpoint: {ckpt_path}")

    state = torch.load(ckpt_path, map_location=device)
    model.load_state_dict(state)
    model.to(device)
    model.eval()
    return model


def collect_stage_accuracies(
    model,
    seeds,
    checkpoint_specs,
    path,
    dpa_timing,
    gng_timing,
    input_size,
    noise,
    readout_index,
    device,
    n_eval_repeats=20,
    n_eval_trials=1024,
):
    """
    Returns
    -------
    accs:
        dict with keys:
            accs["DPA task"][stage] = array of accuracies
            accs["GNG task"][stage] = array of accuracies

        Each array pools seed × repeat accuracies.
    """
    accs = {
        "DPA task": {},
        "GNG task": {},
    }

    for stage, fname in checkpoint_specs.items():
        dpa_vals = []
        gng_vals = []

        for seed in seeds:
            load_checkpoint_for_seed(
                model=model,
                path=path,
                seed=seed,
                filename_template=fname,
                device=device,
            )

            for rep in range(n_eval_repeats):
                dpa_res = eval_dpa_accuracy(
                    model,
                    dpa_timing,
                    n_trials=n_eval_trials,
                    input_size=input_size,
                    noise=noise,
                    readout_index=readout_index,
                    device=device,
                )

                gng_res = eval_gng_accuracy(
                    model,
                    gng_timing,
                    n_trials=n_eval_trials,
                    input_size=input_size,
                    noise=noise,
                    readout_index=readout_index,
                    device=device,
                )

                dpa_vals.append(dpa_res["acc"])
                gng_vals.append(gng_res["acc"])

        accs["DPA task"][stage] = np.asarray(dpa_vals, dtype=float)
        accs["GNG task"][stage] = np.asarray(gng_vals, dtype=float)

    return accs

def mean_sem(x):
    x = np.asarray(x, dtype=float)
    mean = np.nanmean(x)
    if len(x) <= 1:
        sem = 0.0
    else:
        sem = np.nanstd(x, ddof=1) / np.sqrt(np.isfinite(x).sum())
    return mean, sem
